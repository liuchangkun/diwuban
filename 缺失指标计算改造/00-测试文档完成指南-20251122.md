### 方法2：使用脚本自动生成

**创建生成脚本**：`scripts/generate_test_docs.py`

```python
#!/usr/bin/env python3
"""测试文档自动生成脚本"""

import re
from pathlib import Path

def generate_test_docs(source_metric, target_metric, replacements):
    """
    基于源指标生成目标指标的测试文档
    
    Args:
        source_metric: 源指标名称（如'pump_shaft_power'）
        target_metric: 目标指标名称（如'pump_hydraulic_power'）
        replacements: 替换规则字典
    """
    source_dir = Path(f"缺失指标计算改造/{source_metric}")
    target_dir = Path(f"缺失指标计算改造/{target_metric}")
    
    for doc_num in ['06', '07', '08']:
        source_file = source_dir / f"{doc_num}-{source_metric}测试用例.md"
        target_file = target_dir / f"{doc_num}-{target_metric}测试用例.md"
        
        if source_file.exists():
            content = source_file.read_text(encoding='utf-8')
            
            # 应用替换规则
            for old, new in replacements.items():
                content = content.replace(old, new)
            
            target_file.write_text(content, encoding='utf-8')
            print(f"✅ 生成: {target_file}")

# 使用示例
replacements_hydraulic = {
    'pump_shaft_power': 'pump_hydraulic_power',
    'metric_id=66': 'metric_id=67',
    'pump_active_power': 'pump_flow_rate, pump_head',
    'P_shaft': 'P_h',
    '600.0': '500.0',
}

generate_test_docs('pump_shaft_power', 'pump_hydraulic_power', replacements_hydraulic)
```
