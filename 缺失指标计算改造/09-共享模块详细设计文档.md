## 🎯 设计目标

### 核心目标

1. **全局唯一实例**：共享模块在整个系统中只有一个实例
2. **高性能**：支持大数据量的高效处理
3. **可扩展**：支持未来新增指标
4. **易维护**：清晰的接口和职责划分

### 共享模块清单

| 模块 | 职责 | 优先级 |
|------|------|--------|
| **Scheduler** | 任务调度、分片、并行执行 | 高 |
| **DataWriter** | 批量数据写入、自适应批量管理 | 高 |
| **ParameterManager** | 三级参数配置管理 | 高 |
| **SharedServices** | 单例模式管理共享模块 | 高 |
| **Optimizer** | 参数优化（暂不实施） | 低 |

## 📦 模块1：Scheduler（调度器）

### 职责

1. **任务分片**：将大任务拆分为小任务（按设备、按时间）
2. **并行调度**：管理多个任务的并行执行
3. **进度跟踪**：实时跟踪任务进度
4. **错误处理**：处理任务失败和重试

### 类定义

```python
from typing import List, Dict, Callable, Optional
from dataclasses import dataclass
from concurrent.futures import ThreadPoolExecutor, as_completed
import uuid
from datetime import datetime, timedelta

@dataclass
class Task:
    """任务定义"""
    task_id: str
    metric_key: str
    device_id: int
    station_id: int
    start_time: datetime
    end_time: datetime
    priority: int = 100
    retry_count: int = 0
    max_retries: int = 3

@dataclass
class TaskResult:
    """任务结果"""
    task_id: str
    device_id: int
    metric_key: str
    success: bool
    error_message: Optional[str] = None
    duration_ms: int = 0
    results_count: int = 0

class Scheduler:
    """
    任务调度器
    
    职责：
    - 任务分片（按设备、按时间）
    - 并行调度（线程池）
    - 进度跟踪
    - 错误处理和重试
    """
    
    def __init__(self, max_workers: int = 10):
        """
        Args:
            max_workers: 最大并行工作线程数
        """
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        self.active_tasks: Dict[str, Task] = {}
        
        from app.core.logging.setup import log_activity
        self.logger = log_activity
    
    def create_tasks(
        self,
        metric_key: str,
        device_ids: List[int],
        start_time: datetime,
        end_time: datetime,
        time_chunk_hours: int = 1
    ) -> List[Task]:
        """
        创建任务列表（分片）
        
        Args:
            metric_key: 指标键
            device_ids: 设备ID列表
            start_time: 开始时间
            end_time: 结束时间
            time_chunk_hours: 时间分片大小（小时）
        
        Returns:
            任务列表
        """
        tasks = []
        
        # 按设备分片
        for device_id in device_ids:
            # 按时间分片
            current_start = start_time
            while current_start < end_time:
                current_end = min(
                    current_start + timedelta(hours=time_chunk_hours),
                    end_time
                )
                
                task = Task(
                    task_id=str(uuid.uuid4()),
                    metric_key=metric_key,
                    device_id=device_id,
                    station_id=self._get_station_id(device_id),
                    start_time=current_start,
                    end_time=current_end
                )
                tasks.append(task)
                
                current_start = current_end
        
        self.logger(
            f"[调度器] 创建了{len(tasks)}个任务",
            extra={'extra_data': {
                'metric_key': metric_key,
                'device_count': len(device_ids),
                'time_range': [str(start_time), str(end_time)],
                'time_chunk_hours': time_chunk_hours
            }}
        )
        
        return tasks
    
    def _get_station_id(self, device_id: int) -> int:
        """获取设备所属泵站ID"""
        # TODO: 从数据库查询
        return 14  # 临时硬编码

    def execute_tasks(
        self,
        tasks: List[Task],
        calculator_func: Callable[[Task], TaskResult]
    ) -> List[TaskResult]:
        """
        并行执行任务

        Args:
            tasks: 任务列表
            calculator_func: 计算函数（接收Task，返回TaskResult）

        Returns:
            任务结果列表
        """
        results = []
        futures = {}

        # 提交所有任务
        for task in tasks:
            self.active_tasks[task.task_id] = task
            future = self.executor.submit(self._execute_single_task, task, calculator_func)
            futures[future] = task

        # 等待任务完成
        completed_count = 0
        total_count = len(tasks)

        for future in as_completed(futures):
            task = futures[future]
            try:
                result = future.result()
                results.append(result)

                completed_count += 1
                progress = (completed_count / total_count) * 100

                self.logger(
                    f"[调度器] 任务进度: {completed_count}/{total_count} ({progress:.1f}%)",
                    extra={'extra_data': {
                        'task_id': task.task_id,
                        'device_id': task.device_id,
                        'metric_key': task.metric_key,
                        'success': result.success
                    }}
                )

            except Exception as e:
                self.logger(
                    f"[调度器] 任务执行异常: {str(e)}",
                    extra={'extra_data': {
                        'task_id': task.task_id,
                        'device_id': task.device_id,
                        'error': str(e)
                    }}
                )
                results.append(TaskResult(
                    task_id=task.task_id,
                    device_id=task.device_id,
                    metric_key=task.metric_key,
                    success=False,
                    error_message=str(e)
                ))
            finally:
                del self.active_tasks[task.task_id]

        return results

    def _execute_single_task(
        self,
        task: Task,
        calculator_func: Callable[[Task], TaskResult]
    ) -> TaskResult:
        """
        执行单个任务（带重试）

        Args:
            task: 任务
            calculator_func: 计算函数

        Returns:
            任务结果
        """
        import time

        for attempt in range(task.max_retries + 1):
            try:
                start_time = time.time()
                result = calculator_func(task)
                duration_ms = int((time.time() - start_time) * 1000)
                result.duration_ms = duration_ms
                return result

            except Exception as e:
                if attempt < task.max_retries:
                    self.logger(
                        f"[调度器] 任务失败，重试 {attempt + 1}/{task.max_retries}",
                        extra={'extra_data': {
                            'task_id': task.task_id,
                            'device_id': task.device_id,
                            'error': str(e)
                        }}
                    )
                    time.sleep(2 ** attempt)  # 指数退避
                else:
                    raise

    def shutdown(self):
        """关闭调度器"""
        self.executor.shutdown(wait=True)
        self.logger("[调度器] 已关闭")
```

## 📝 模块2：DataWriter（数据写入器）

### 职责

1. **批量写入**：将计算结果批量写入数据库
2. **自适应批量**：根据数据量自动调整批量大小
3. **错误处理**：处理写入失败和重试
4. **性能监控**：记录写入性能指标

### 类定义

```python
from typing import List, Dict
import time
from dataclasses import dataclass

@dataclass
class WriteRecord:
    """写入记录"""
    device_id: int
    metric_key: str
    timestamp: datetime
    value: float
    quality_code: int = 0

class DataWriter:
    """
    数据写入器

    职责：
    - 批量写入fact_measurements
    - 自适应批量大小
    - 错误处理和重试
    - 性能监控
    """

    def __init__(
        self,
        initial_batch_size: int = 1000,
        min_batch_size: int = 100,
        max_batch_size: int = 10000,
        target_duration_ms: int = 500
    ):
        """
        Args:
            initial_batch_size: 初始批量大小
            min_batch_size: 最小批量大小
            max_batch_size: 最大批量大小
            target_duration_ms: 目标写入耗时（毫秒）
        """
        self.batch_size = initial_batch_size
        self.min_batch_size = min_batch_size
        self.max_batch_size = max_batch_size
        self.target_duration_ms = target_duration_ms

        self.total_written = 0
        self.total_duration_ms = 0

        from app.core.logging.setup import log_activity
        from app.core.database.connection import execute_batch
        self.logger = log_activity
        self.execute_batch = execute_batch

    def write(self, records: List[WriteRecord]) -> int:
        """
        批量写入记录

        Args:
            records: 写入记录列表

        Returns:
            成功写入的记录数
        """
        if not records:
            return 0

        total_written = 0

        # 分批写入
        for i in range(0, len(records), self.batch_size):
            batch = records[i:i + self.batch_size]
            written = self._write_batch(batch)
            total_written += written

        self.logger(
            f"[数据写入] 写入完成: {total_written}条记录",
            extra={'extra_data': {
                'total_records': len(records),
                'batch_size': self.batch_size,
                'avg_duration_ms': self.total_duration_ms / max(1, total_written / self.batch_size)
            }}
        )

        return total_written

    def _write_batch(self, batch: List[WriteRecord]) -> int:
        """
        写入单个批次

        Args:
            batch: 批次记录

        Returns:
            成功写入的记录数
        """
        start_time = time.time()

        try:
            # 构造SQL
            sql = """
                INSERT INTO fact_measurements (device_id, metric_key, timestamp, value, quality_code)
                VALUES (%s, %s, %s, %s, %s)
                ON CONFLICT (device_id, metric_key, timestamp)
                DO UPDATE SET value = EXCLUDED.value, quality_code = EXCLUDED.quality_code, updated_at = NOW()
            """

            # 准备数据
            data = [
                (r.device_id, r.metric_key, r.timestamp, r.value, r.quality_code)
                for r in batch
            ]

            # 批量执行
            self.execute_batch(sql, data)

            # 记录性能
            duration_ms = int((time.time() - start_time) * 1000)
            self.total_written += len(batch)
            self.total_duration_ms += duration_ms

            # 自适应调整批量大小
            self._adjust_batch_size(duration_ms)

            return len(batch)

        except Exception as e:
            self.logger(
                f"[数据写入] 批次写入失败: {str(e)}",
                extra={'extra_data': {
                    'batch_size': len(batch),
                    'error': str(e)
                }}
            )
            raise

    def _adjust_batch_size(self, duration_ms: int):
        """
        自适应调整批量大小

        目标：保持写入耗时在target_duration_ms附近

        Args:
            duration_ms: 本次写入耗时
        """
        if duration_ms < self.target_duration_ms * 0.5:
            # 耗时太短，增加批量大小
            new_batch_size = int(self.batch_size * 1.5)
            self.batch_size = min(new_batch_size, self.max_batch_size)

        elif duration_ms > self.target_duration_ms * 2:
            # 耗时太长，减少批量大小
            new_batch_size = int(self.batch_size * 0.7)
            self.batch_size = max(new_batch_size, self.min_batch_size)

        # 记录调整
        if self.batch_size != self.batch_size:
            self.logger(
                f"[数据写入] 批量大小调整: {self.batch_size} → {self.batch_size}",
                extra={'extra_data': {
                    'duration_ms': duration_ms,
                    'target_duration_ms': self.target_duration_ms
                }}
            )

    def get_stats(self) -> Dict:
        """获取写入统计"""
        return {
            'total_written': self.total_written,
            'total_duration_ms': self.total_duration_ms,
            'avg_duration_ms': self.total_duration_ms / max(1, self.total_written / self.batch_size),
            'current_batch_size': self.batch_size
        }
```

## 📋 模块3：ParameterManager（参数管理器）

### 职责

1. **三级配置加载**：全局 > 泵站 > 设备
2. **参数缓存**：避免重复查询数据库
3. **参数更新**：支持动态更新参数
4. **默认值管理**：提供合理的默认值

### 类定义

```python
from typing import Dict, Optional
from functools import lru_cache

class ParameterManager:
    """
    参数管理器

    职责：
    - 三级参数配置加载（全局 > 泵站 > 设备）
    - 参数缓存
    - 默认值管理
    """

    def __init__(self):
        """初始化参数管理器"""
        from app.core.logging.setup import log_activity
        from app.core.database.connection import query
        self.logger = log_activity
        self.query = query

        # 默认参数
        self.default_params = {
            'pump_flow_rate': {
                'method_a': {'alpha': 1.0, 'beta': 1.0},
                'method_b': {'smooth_window': 5},
                'method_c': {},
                'method_d': {'alpha': 1.0},
                'method_e': {'beta': 1.0},
                'method_f': {'model_type': 'linear', 'min_samples': 100},
                'global': {
                    'max_flow': 500.0,
                    'max_power': 200.0,
                    'max_freq': 50.0
                }
            }
        }

        # 参数缓存
        self._cache = {}

    def get_params(
        self,
        metric_key: str,
        method_id: Optional[str] = None,
        station_id: Optional[int] = None,
        device_id: Optional[int] = None
    ) -> Dict[str, float]:
        """
        获取参数（三级优先级：设备 > 泵站 > 全局）

        Args:
            metric_key: 指标键
            method_id: 方法ID（None表示全局参数）
            station_id: 泵站ID（None表示全局）
            device_id: 设备ID（None表示泵站级）

        Returns:
            参数字典
        """
        # 构造缓存键
        cache_key = f"{metric_key}:{method_id}:{station_id}:{device_id}"

        # 检查缓存
        if cache_key in self._cache:
            return self._cache[cache_key]

        # 从数据库加载
        params = self._load_params_from_db(metric_key, method_id, station_id, device_id)

        # 如果数据库没有，使用默认值
        if not params:
            params = self._get_default_params(metric_key, method_id)

        # 缓存
        self._cache[cache_key] = params

        self.logger(
            f"[参数管理] 加载参数",
            extra={'extra_data': {
                'metric_key': metric_key,
                'method_id': method_id,
                'station_id': station_id,
                'device_id': device_id,
                'params': params,
                'from_cache': cache_key in self._cache
            }}
        )

        return params

    def _load_params_from_db(
        self,
        metric_key: str,
        method_id: Optional[str],
        station_id: Optional[int],
        device_id: Optional[int]
    ) -> Dict[str, float]:
        """
        从数据库加载参数（三级优先级）

        Args:
            metric_key: 指标键
            method_id: 方法ID
            station_id: 泵站ID
            device_id: 设备ID

        Returns:
            参数字典
        """
        sql = """
            WITH param_hierarchy AS (
                SELECT
                    parameter_name,
                    parameter_value,
                    CASE
                        WHEN device_id IS NOT NULL THEN 1  -- 设备级
                        WHEN station_id IS NOT NULL THEN 2  -- 泵站级
                        ELSE 3  -- 全局级
                    END AS priority
                FROM calculation_parameters
                WHERE metric_key = %s
                  AND (method_id = %s OR method_id IS NULL)
                  AND (
                      (station_id = %s AND device_id = %s) OR  -- 设备级
                      (station_id = %s AND device_id IS NULL) OR  -- 泵站级
                      (station_id IS NULL AND device_id IS NULL)  -- 全局级
                  )
            )
            SELECT DISTINCT ON (parameter_name) parameter_name, parameter_value
            FROM param_hierarchy
            ORDER BY parameter_name, priority
        """

        rows = self.query(sql, [
            metric_key,
            method_id,
            station_id, device_id,
            station_id
        ])

        # 转换为字典
        params = {row['parameter_name']: row['parameter_value'] for row in rows}

        return params

    def _get_default_params(self, metric_key: str, method_id: Optional[str]) -> Dict[str, float]:
        """
        获取默认参数

        Args:
            metric_key: 指标键
            method_id: 方法ID

        Returns:
            默认参数字典
        """
        if metric_key not in self.default_params:
            return {}

        if method_id is None:
            method_id = 'global'

        return self.default_params[metric_key].get(method_id, {}).copy()

    def update_params(
        self,
        metric_key: str,
        method_id: str,
        params: Dict[str, float],
        station_id: Optional[int] = None,
        device_id: Optional[int] = None
    ):
        """
        更新参数（写入数据库并清除缓存）

        Args:
            metric_key: 指标键
            method_id: 方法ID
            params: 参数字典
            station_id: 泵站ID
            device_id: 设备ID
        """
        from app.core.database.connection import execute

        for param_name, param_value in params.items():
            sql = """
                INSERT INTO calculation_parameters (
                    metric_key, method_id, parameter_name, parameter_value,
                    station_id, device_id
                )
                VALUES (%s, %s, %s, %s, %s, %s)
                ON CONFLICT (metric_key, method_id, parameter_name, station_id, device_id)
                DO UPDATE SET parameter_value = EXCLUDED.parameter_value, updated_at = NOW()
            """

            execute(sql, [
                metric_key, method_id, param_name, param_value,
                station_id, device_id
            ])

        # 清除缓存
        self.clear_cache()

        self.logger(
            f"[参数管理] 更新参数",
            extra={'extra_data': {
                'metric_key': metric_key,
                'method_id': method_id,
                'params': params,
                'station_id': station_id,
                'device_id': device_id
            }}
        )

    def clear_cache(self):
        """清除参数缓存"""
        self._cache.clear()
        self.logger("[参数管理] 缓存已清除")

## 🔧 模块4：SharedServices（服务管理器）

### 职责

1. **单例模式**：确保共享模块全局唯一
2. **生命周期管理**：初始化和销毁
3. **依赖注入**：为指标模块提供共享服务

### 类定义

```python
from typing import Optional

class SharedServices:
    """
    共享服务管理器（单例模式）

    职责：
    - 管理共享模块的生命周期
    - 确保全局唯一实例
    - 提供依赖注入
    """

    _instance: Optional['SharedServices'] = None
    _initialized: bool = False

    def __new__(cls):
        """单例模式"""
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self):
        """初始化（只执行一次）"""
        if SharedServices._initialized:
            return

        from app.core.logging.setup import log_activity
        self.logger = log_activity

        # 初始化共享模块
        self.scheduler = Scheduler(max_workers=10)
        self.data_writer = DataWriter(
            initial_batch_size=1000,
            target_duration_ms=500
        )
        self.parameter_manager = ParameterManager()

        SharedServices._initialized = True

        self.logger(
            "[共享服务] 初始化完成",
            extra={'extra_data': {
                'scheduler_workers': self.scheduler.max_workers,
                'data_writer_batch_size': self.data_writer.batch_size
            }}
        )

    @classmethod
    def get_instance(cls) -> 'SharedServices':
        """获取单例实例"""
        if cls._instance is None:
            cls._instance = cls()
        return cls._instance

    def shutdown(self):
        """关闭所有服务"""
        self.scheduler.shutdown()
        self.logger("[共享服务] 已关闭")
```

## 📊 使用示例

### 示例1：完整计算流程

```python
from datetime import datetime, timedelta

# 1. 获取共享服务
services = SharedServices.get_instance()
scheduler = services.scheduler
data_writer = services.data_writer
param_manager = services.parameter_manager

# 2. 创建任务
device_ids = [105, 106, 107]
start_time = datetime(2025, 1, 1, 0, 0, 0)
end_time = datetime(2025, 1, 1, 1, 0, 0)

tasks = scheduler.create_tasks(
    metric_key='pump_flow_rate',
    device_ids=device_ids,
    start_time=start_time,
    end_time=end_time,
    time_chunk_hours=1
)

# 3. 定义计算函数
def calculate_pump_flow_rate(task: Task) -> TaskResult:
    """计算pump_flow_rate"""
    from app.services.calculation.metrics.pump_flow_rate import PumpFlowRatePipeline

    # 创建Pipeline
    pipeline = PumpFlowRatePipeline(
        device_id=task.device_id,
        start_time=task.start_time,
        end_time=task.end_time,
        param_manager=param_manager
    )

    # 执行计算
    results = pipeline.execute()

    # 写入结果
    write_records = [
        WriteRecord(
            device_id=task.device_id,
            metric_key='pump_flow_rate',
            timestamp=r['timestamp'],
            value=r['value']
        )
        for r in results
    ]
    written_count = data_writer.write(write_records)

    return TaskResult(
        task_id=task.task_id,
        device_id=task.device_id,
        metric_key='pump_flow_rate',
        success=True,
        results_count=written_count
    )

# 4. 执行任务
results = scheduler.execute_tasks(tasks, calculate_pump_flow_rate)

# 5. 统计结果
success_count = sum(1 for r in results if r.success)
total_count = len(results)
print(f"完成: {success_count}/{total_count}")
```

def create_tasks_adaptive(
    self,
    metric_key: str,
    device_ids: List[int],
    start_time: datetime,
    end_time: datetime

    """自适应任务分片"""

    # 估算数据量
    total_hours = (end_time - start_time).total_seconds() / 3600
    estimated_records = len(device_ids) * total_hours * 3600  # 1秒1条

    # 根据数据量调整分片大小
    if estimated_records < 100000:
        time_chunk_hours = 24  # 小数据量，大分片
    elif estimated_records < 1000000:
        time_chunk_hours = 6   # 中等数据量，中等分片
    else:
        time_chunk_hours = 1   # 大数据量，小分片

    return self.create_tasks(
        metric_key, device_ids, start_time, end_time, time_chunk_hours
    )

### 2. DataWriter优化

**优化点1：异步写入**
```python
import asyncio
from asyncio import Queue

class AsyncDataWriter(DataWriter):
    """异步数据写入器"""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.write_queue = Queue(maxsize=10000)
        self.writer_task = None

    async def start(self):
        """启动异步写入任务"""
        self.writer_task = asyncio.create_task(self._writer_loop())

    async def _writer_loop(self):
        """写入循环"""
        batch = []

        while True:
            try:
                # 等待记录或超时
                record = await asyncio.wait_for(
                    self.write_queue.get(),
                    timeout=1.0
                )
                batch.append(record)

                # 达到批量大小，立即写入
                if len(batch) >= self.batch_size:
                    self._write_batch(batch)
                    batch = []

            except asyncio.TimeoutError:
                # 超时，写入当前批次
                if batch:
                    self._write_batch(batch)
                    batch = []

    async def write_async(self, records: List[WriteRecord]):
        """异步写入"""
        for record in records:
            await self.write_queue.put(record)
```

import io
import csv

def _write_batch_copy(self, batch: List[WriteRecord]) -> int:
    """使用COPY命令批量写入（更快）"""
