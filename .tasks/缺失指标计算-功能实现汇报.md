**步骤3：拓扑排序（Kahn算法）**
```python
def topological_sort(graph):
    # 计算入度
    in_degree = {node: 0 for node in graph}
    for node in graph:
        for dep in graph[node]:
            if dep in in_degree:
                in_degree[dep] += 1
    
    # 找到入度为0的节点
    queue = [node for node in in_degree if in_degree[node] == 0]
    result = []
    
    # Kahn算法
    while queue:
        node = queue.pop(0)
        result.append(node)
        for neighbor in graph:
            if node in graph[neighbor]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)
    
    return result
```
- **输入**: 依赖图
- **输出**: 计算顺序列表
- **时间复杂度**: O(V+E)

**步骤4：检测循环依赖（DFS算法）**
```python
def detect_circular_dependency(graph):
    visited = set()
    rec_stack = set()
    
    def dfs(node):
        visited.add(node)
        rec_stack.add(node)
        
        for neighbor in graph.get(node, []):
            if neighbor not in visited:
                if dfs(neighbor):
                    return True
            elif neighbor in rec_stack:
                return True  # 发现循环
        
        rec_stack.remove(node)
        return False
    
    for node in graph:
        if node not in visited:
            if dfs(node):
                return True
    
    return False
```
- **输入**: 依赖图
- **输出**: True（有循环）或 False（无循环）
- **时间复杂度**: O(V+E)

```python
def check_dependencies(dependencies, available_metrics, data=None):
    # 检查依赖指标是否在可用列表中
    for dep in dependencies:
        if dep not in available_metrics:
            return False
    
    # 如果提供了data，检查数据有效性
    if data is not None:
        for dep in dependencies:
            if dep not in data:
                return False
            
            values = data[dep]
            # 检查是否全是NaN
            if np.all(np.isnan(values)):
                return False
            
            # 对于某些指标，检查是否有非零值
            if dep in ['pump_frequency', 'pump_flow_rate']:
                if np.all(values[~np.isnan(values)] == 0):
                    return False
    
    return True
```
- **输入**: 依赖列表, 可用指标列表, 数据字典（可选）
- **输出**: True（满足）或 False（不满足）
- **作用**: 确保依赖数据可用且有效

```python
    methods = _method_registry.get(metric_key, [])

    for method in methods:
```

```python
def calculate_xxx_method_y(
    data: Dict[str, np.ndarray],
    params: Dict[str, float]
) -> np.ndarray:
    """
    方法Y：方法名称
    
    公式: result = f(input1, input2, ...)
    
    Args:
        data: 包含 input1, input2, ...
        params: 包含 param1, param2, ...
    
    Returns:
        计算得到的结果数组
    """
    # 1. 提取输入数据
    input1 = data['input1']
    input2 = data['input2']
    
    # 2. 提取参数（带默认值）
    param1 = params.get('param1', default_value)
    
    # 3. 创建有效数据掩码
    valid_mask = ~np.isnan(input1) & ~np.isnan(input2) & (input1 > 0)
    
    # 4. 初始化结果数组为NaN
    result = np.full_like(input1, np.nan)
    
    # 5. 对有效数据进行计算
    result[valid_mask] = formula(input1[valid_mask], input2[valid_mask], param1)
    
    # 6. 返回结果
    return result
```

```python
def calculate_pump_speed_method_a(
    data: Dict[str, np.ndarray],
    params: Dict[str, float]
) -> np.ndarray:
    """方法A：相对转速法"""
    # 1. 提取输入数据
    f = data['pump_frequency']  # Hz
    
    # 2. 提取参数
    f_ref = params.get('f_ref', 50.0)  # Hz
    
    # 3. 创建有效数据掩码

    # 4. 初始化结果数组为NaN
    n_star = np.full_like(f, np.nan)
    
    # 5. 计算相对转速
    n_star[valid_mask] = f[valid_mask] / f_ref
    
    # 6. 返回结果
    return n_star
```

```python
def validate(metric_key, values, data=None, context=None):
    validators = _validation_config.get(metric_key, [])
    
    all_valid = np.ones(len(values), dtype=bool)
    errors = []
    
    for validator_config in validators:
        validator_type = validator_config['validator_type']
        params = validator_config['params']
        
        if validator_type == 'range':
            mask, errs = validate_range(values, params['min'], params['max'])
        elif validator_type == 'non_negative':
            mask, errs = validate_non_negative(values)
        elif validator_type == 'not_nan':
            mask, errs = validate_not_nan(values)
        
        all_valid &= mask
        errors.extend(errs)
    
    return all_valid, errors
```
