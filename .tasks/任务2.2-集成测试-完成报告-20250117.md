### 示例1：测试阶段1执行顺序
```python
@patch("app.services.ingest.prepare_dim.get_conn")
@patch("app.services.ingest.prepare_dim._backup_manual_tables")
@patch("app.services.ingest.prepare_dim._clear_dependent_tables")
@patch("app.services.ingest.prepare_dim._restore_manual_tables")
@patch("app.services.ingest.prepare_dim._rebuild_config_tables")
@patch("app.services.ingest.prepare_dim._ensure_sequences")
@patch("app.services.ingest.prepare_dim._reload_metric_config_from_sql")
def test_stage1_executes_correct_steps(
    self,
    mock_reload_metric,
    mock_ensure_seq,
    mock_rebuild_config,
    mock_restore,
    mock_clear,
    mock_backup,
    mock_get_conn,
    mock_settings,
    mock_mapping_file,
):
    """测试阶段1执行正确的步骤顺序"""
    # Mock 数据库连接
    mock_conn = MagicMock()
    mock_cur = MagicMock()
    mock_conn.cursor.return_value.__enter__.return_value = mock_cur
    mock_get_conn.return_value.__enter__.return_value = mock_conn
    
    # Mock 备份数据
    mock_backup.return_value = {
        "dim_device_capabilities": [(1, "站点A", "设备1", "key1", "value1", "备注1")],
        "dim_metric_metadata_override": [],
    }
    
    # Mock 清空表返回值
    mock_clear.return_value = 100
    
    # Mock 指标数量查询
    mock_cur.fetchone.return_value = [50]
    
    # 执行阶段1
    result = prepare_dim(mock_settings, mock_mapping_file, stage=1)
    
    # 验证返回值
    assert result["success"] is True
    assert result["stage"] == 1
    assert result["stations_count"] == 2
    assert result["devices_count"] == 3
    assert result["metrics_count"] == 50
    
    # 验证执行顺序
    mock_backup.assert_called_once_with(mock_cur)
    mock_clear.assert_called_once_with(mock_cur)
    mock_restore.assert_called_once()
    mock_rebuild_config.assert_called_once_with(mock_settings, mock_cur)
    
    # 验证提交事务
    mock_conn.commit.assert_called()
```
