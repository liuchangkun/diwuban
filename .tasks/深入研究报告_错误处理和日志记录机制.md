def init_logging(
    config_dir: Path | str = "configs", system_tz: str = "Asia/Shanghai"
) -> None:
    """初始化日志系统。重复调用将重置全局 logging 配置。"""
    global _global_async_mgr
    
    # 修复 Windows 终端中文乱码：强制 stdout/stderr 使用 UTF-8
    try:
        if hasattr(sys.stdout, "reconfigure"):
            sys.stdout.reconfigure(encoding="utf-8", errors="replace")
        if hasattr(sys.stderr, "reconfigure"):
            sys.stderr.reconfigure(encoding="utf-8", errors="replace")
    except Exception:
        pass
    
    config_dir = Path(config_dir)
    log_yaml = config_dir / "logging.yaml"
    cfg = LoggingConfig.from_yaml(log_yaml, system_tz)
    
    # 根 logger 基础级别
    root_level = getattr(logging, cfg.level.upper(), logging.INFO)
    logging.root.handlers.clear()
    logging.root.setLevel(root_level)
    
    # 绑定 handlers（异步/同步两种模式）
    if cfg.async_cfg.enabled:
        _global_async_mgr = AsyncLoggingManager(cfg)
        _global_async_mgr.start(handlers)
        logging.root.addHandler(_global_async_mgr.queue_handler)
        # 进程退出时确保 flush 队列，避免丢日志
        import atexit as _atexit
        _atexit.register(lambda: (_global_async_mgr and _global_async_mgr.stop()))
    else:
        for h in handlers:
            logging.root.addHandler(h)
    
    # 安装未捕获异常钩子
    _install_exception_hooks()
    
    # 热加载监视
    _start_hot_reload_if_needed(cfg, log_yaml, cfg.timezone)
```

**关键特性**：
- ✅ 支持异步日志（避免阻塞）
- ✅ 支持热加载（无需重启）
- ✅ 支持多种输出格式（pipe、json、keyvalue）
- ✅ 支持时区本地化

def log_sql(
    sql: str,
    params: Optional[dict] = None,
    duration_ms: Optional[float] = None,
    rows: Optional[int] = None,
    error: Optional[str] = None,
    level: int = logging.INFO,
    slow_threshold: Optional[int] = None,
    thread_name: Optional[str] = None,
) -> None:
    """
    记录SQL执行日志
    
    参数：
        sql: SQL语句
        params: SQL参数
        duration_ms: 执行耗时（毫秒）
        rows: 影响行数
        error: 错误信息
        level: 日志级别
        slow_threshold: 慢SQL阈值（毫秒）
        thread_name: 线程名称
    """
    logger = logging.getLogger("sql")
    
    extra = {
        "type": "SQL",
        "sql": sql,
        "params": params or {},
        "duration_ms": duration_ms,
        "rows": rows,
        "result": ("error" if error else "ok"),
    }
    if thread_name:
        extra["threadName"] = thread_name
    if error:
        extra["error"] = error
    
    # 输出日志（含慢SQL标记）
    slow_flag = False
    if slow_threshold and duration_ms is not None:
        try:
            slow_flag = duration_ms >= int(slow_threshold)
        except Exception:
            slow_flag = False
    
    logger.log(
        level,
        "[SQL] 执行",
        extra={
            "extra_data": {
                **extra,
                **({"slow": True} if slow_flag else {}),
            }
        },
    )
```

**关键特性**：
- ✅ 记录SQL语句和参数
- ✅ 记录执行耗时和影响行数
- ✅ 标记慢SQL（超过阈值）
- ✅ 记录错误信息

### call_with_activity - 活动日志

**代码位置**：`app/core/logging/setup.py:955-985`

```python
def call_with_activity(name: str, func: Callable, *args, **kwargs) -> Any:
    """
    调用函数并记录活动日志（入口、退出、错误）
    
    参数：
        name: 活动名称
        func: 要调用的函数
        *args, **kwargs: 函数参数
    
    返回：
        函数返回值
    """
    logger = logging.getLogger(__name__)
    start = datetime.now()
    
    try:
        result = func(*args, **kwargs)
        
        # 退出日志：慢标记 + 链路携带
        dur_ms = int((datetime.now() - start).total_seconds() * 1000)
        slow_flag = False
        try:
            if _global_cfg and _global_cfg.activity_step_slow_ms > 0:
                slow_flag = dur_ms >= _global_cfg.activity_step_slow_ms
        except Exception:
            pass
        logger.info(
            "[退出] %s",
            name,
            extra={
                "extra_data": {
                    "类型": "退出",
                    "event": "exit",
                    "duration_ms": dur_ms,
                    **({"slow": True} if slow_flag else {}),
                }
            },
        )
        return result
    except Exception:
        dur_ms = int((datetime.now() - start).total_seconds() * 1000)
        logger.error(
            "[错误] %s",
            name,
            extra={
                "extra_data": {"类型": "错误", "event": "fail", "duration_ms": dur_ms}
            },
            exc_info=True,
        )
        raise
```

**关键特性**：
- ✅ 自动记录入口、退出、错误
- ✅ 记录执行耗时
- ✅ 标记慢操作
- ✅ 记录异常堆栈

## 🔥 错误处理机制

### error_handler装饰器

**代码位置**：`app/core/exceptions.py:331-406`

```python
def error_handler(
    logger_name: Optional[str] = None,
    log_level: int = 40,
    reraise: bool = True,
    context_fields: Optional[list[str]] = None,
) -> Callable[[F], F]:
    """
    统一错误处理装饰器
    
    功能：
    - 自动记录异常信息到日志
    - 提供结构化的上下文信息
    - 支持异常转换和重新抛出
    - 支持自定义日志记录器和级别
    
    参数：
        logger_name: 自定义日志记录器名称，默认使用被装饰函数的模块名
        log_level: 日志级别，默认为 ERROR
        reraise: 是否重新抛出异常，默认为 True
        context_fields: 从函数参数中提取的上下文字段列表
    
    使用示例：
        @error_handler(context_fields=['file_path', 'station_id'])
        def process_file(file_path: str, station_id: int) -> None:
            # 函数实现
            pass
    """
    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 构建上下文信息
            context = {
                "function": func.__name__,
                "module": func.__module__,
            }
            
            # 提取指定的上下文字段
            if context_fields:
                import inspect
                sig = inspect.signature(func)
                bound_args = sig.bind_partial(*args, **kwargs)
                bound_args.apply_defaults()
                
                for field in context_fields:
                    if field in bound_args.arguments:
                        context[field] = bound_args.arguments[field]
            
            try:
                return func(*args, **kwargs)
            except BaseAppException as e:
                # 应用程序异常，已经结构化，直接记录
                context.update(e.context)
                if reraise:
                    raise
            
            except Exception as e:
                # 未预期的异常，包装为应用异常
                if reraise:
                    raise BaseAppException(
                        f"函数 {func.__name__} 执行失败: {str(e)}",
                        error_code="UNEXPECTED_ERROR",
                        context=context,
                        cause=e,
                    ) from e
        
        return wrapper
    
    return decorator
```

### ErrorAnalyzer - 错误分析器

**代码位置**：`app/core/error_analysis.py:142-171`

```python
def record_error(
    self,
    error: Exception,
    context: Dict[str, Any],
    function_name: str = "unknown",
    module_name: str = "unknown",
):
    """记录错误"""
    # 提取错误信息
    if isinstance(error, BaseAppException):
        severity = error.severity
        recovery_action = error.recovery_action.value
        error_context = {**context, **error.context}
    else:
        severity = ErrorSeverity.MEDIUM
        recovery_action = "retry"
        error_context = context
    
    # 创建错误记录
    record = ErrorRecord(
        timestamp=time.time(),
        error_type=type(error).__name__,
        error_message=str(error),
        severity=severity,
        context=error_context,
        function_name=function_name,
        module_name=module_name,
        recovery_action=recovery_action,
    )
    
    # 添加到错误历史
    self._error_history.append(record)
```

## 📌 总结

### 日志系统的优势

| 优势 | 描述 |
|------|------|
| 异步日志 | 避免阻塞主线程 |
| 热加载 | 无需重启即可更新配置 |
| 多种格式 | 支持pipe、json、keyvalue |
| 慢操作标记 | 自动标记慢SQL和慢操作 |
| 脱敏处理 | 自动脱敏敏感字段 |

### 错误处理的关键点

1. **统一装饰器**：error_handler提供统一的错误处理
2. **结构化上下文**：自动提取函数参数作为上下文
3. **错误分析**：ErrorAnalyzer记录和分析错误
4. **异常转换**：将未预期异常包装为应用异常

### 日志记录的最佳实践

1. **使用专用API**：log_sql、log_biz、call_with_activity
2. **记录上下文**：使用extra_data记录结构化信息
3. **标记慢操作**：自动标记超过阈值的操作
4. **记录异常堆栈**：使用exc_info=True记录完整堆栈
