### 2. 增量备份策略

**变化检测算法**（两阶段）:

```python
def _table_has_changed(self, cur, table_name: str) -> bool:
    """
    检测表是否发生变化
    
    策略：
    1. 检查行数是否变化（快速检测）
    2. 如果行数相同，计算整表数据的MD5（精确检测）
    """
    # 阶段1：行数检测
    cur.execute(f"SELECT COUNT(*) FROM {table_name}")
    current_row_count = cur.fetchone()[0]
    
    # 从最新备份文件中提取行数
    match = re.search(r'-- 行数: (\d+)', content)
    backup_row_count = int(match.group(1))
    
    if current_row_count != backup_row_count:
        return True  # 行数变化，需要备份
    
    # 阶段2：MD5检测
    cur.execute(f"SELECT md5(CAST((array_agg(t.* ORDER BY (t.*)::text))AS text)) FROM {table_name} t")
    current_md5 = cur.fetchone()[0]
    
    match = re.search(r'-- MD5: ([a-f0-9]+)', content)
    backup_md5 = match.group(1)
    
    return current_md5 != backup_md5  # MD5不同，需要备份
```

**性能优化**:
- ✅ 行数检测：O(1) 复杂度（使用统计信息）
- ✅ MD5计算：O(n) 复杂度（仅在行数相同时执行）
- ✅ 跳过未变化的表：节省磁盘空间和I/O

**核心逻辑**:
```python
def _generate_sql_backup(self, cur, table_name: str) -> str:
    # 步骤1：获取表结构
    cur.execute("""
        SELECT column_name, data_type
        FROM information_schema.columns
        WHERE table_schema = 'public' AND table_name = %s
        ORDER BY ordinal_position
    """, (table_name,))
    columns = cur.fetchall()
    
    # 步骤2：获取主键
    cur.execute("""
        SELECT a.attname
        FROM pg_index i
        JOIN pg_attribute a ON a.attrelid = i.indrelid AND a.attnum = ANY(i.indkey)
        WHERE i.indrelid = %s::regclass AND i.indisprimary
    """, (table_name,))
    primary_keys = [row[0] for row in cur.fetchall()]
    
    # 步骤3：获取数据
    cur.execute(f"SELECT * FROM {table_name}")
    rows = cur.fetchall()
    
    # 步骤4：计算MD5
    cur.execute(f"SELECT md5(CAST((array_agg(t.* ORDER BY (t.*)::text))AS text)) FROM {table_name} t")
    table_md5 = cur.fetchone()[0]
    
    # 步骤5：生成INSERT语句
    for row in rows:
        values = []
        for col_name, val in zip(column_names, row):
            if val is None:
                values.append("NULL")
            elif isinstance(val, bool):
                values.append("TRUE" if val else "FALSE")
            elif isinstance(val, (int, float)):
                values.append(str(val))
            elif data_type in ('timestamp...', 'date', 'time...'):
                escaped = str(val).replace("'", "''")
                values.append(f"'{escaped}'")
            elif isinstance(val, (list, dict)):
                escaped = json.dumps(val).replace("'", "''")
                values.append(f"'{escaped}'")
            else:
                escaped = str(val).replace("'", "''")
                values.append(f"'{escaped}'")
        
        # 生成ON CONFLICT子句
        if primary_keys:
            pk_list = ", ".join(primary_keys)
            update_set = ", ".join([
                f"{col} = EXCLUDED.{col}"
                for col in column_names
                if col not in primary_keys
            ])
            on_conflict_clause = f" ON CONFLICT ({pk_list}) DO UPDATE SET {update_set}"
        
        sql_lines.append(f"INSERT INTO {table_name} ({column_list}) VALUES ({value_list}){on_conflict_clause};")
```

**类型处理**:
| 数据类型 | 处理方式 | 示例 |
|---------|---------|------|
| NULL | 直接输出 | `NULL` |
| Boolean | TRUE/FALSE | `TRUE` |
| 数值 | 直接转字符串 | `50`, `0.78` |
| 时间戳 | 加引号+转义 | `'2025-10-31 02:28:00+08:00'` |
| JSONB | JSON序列化+转义 | `'{"min": 0, "max": 100}'` |
| 字符串 | 转义单引号 | `'pump''s efficiency'` |

### 4. 版本管理

**版本号生成**:
```python
def _get_next_version(self, table_name: str) -> int:
    # 提取所有版本号
    versions = []
    for f in backup_files:
        match = re.search(r'_v(\d+)_', f.name)
        if match:
            versions.append(int(match.group(1)))
    
    return max(versions) + 1 if versions else 1
```

**旧版本清理**:
```python
def _cleanup_old_versions(self, table_name: str) -> None:
    # 按修改时间排序
    backup_files = sorted(
        table_backup_dir.glob(f"{table_name}_v*.sql"),
        key=lambda x: x.stat().st_mtime,
        reverse=True
    )
    
    # 删除超出保留数量的备份（默认保留60个）
    files_to_remove = backup_files[self.keep_versions:]
    for f in files_to_remove:
        f.unlink()
```

### 2. 恢复算法

**核心逻辑**:
```python
def restore_table(self, cur, table_name: str, version: Optional[int] = None, 
                 target_table: Optional[str] = None) -> Dict[str, Any]:
    # 步骤1：查找备份文件
    backup_files = sorted(
        table_backup_dir.glob(f"{table_name}_v*.sql"),
        key=lambda x: x.stat().st_mtime,
        reverse=True
    )
    
    # 步骤2：选择版本
    if version is None:
        backup_file = backup_files[0]  # 最新版本
    else:
        for f in backup_files:
            if f"_v{version}_" in f.name:
                backup_file = f
                break
    
    # 步骤3：读取并执行SQL
    sql_content = backup_file.read_text(encoding="utf-8")
    
    # 步骤4：表名替换（可选）
    if target_table and target_table != table_name:
        sql_content = sql_content.replace(f"INSERT INTO {table_name}", f"INSERT INTO {target_table}")
    
    # 步骤5：执行SQL
    cur.execute(sql_content)
    
    return {"status": "ok", "version": actual_version, "file": str(backup_file)}
```
