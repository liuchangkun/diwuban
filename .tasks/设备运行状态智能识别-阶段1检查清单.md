SELECT COUNT(*) AS orphan_count
FROM public.device_running_thresholds drt
LEFT JOIN public.dim_devices d ON d.id = drt.device_id
WHERE d.id IS NULL;

-- 如果orphan_count = 0，执行以下SQL
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM information_schema.table_constraints
        WHERE constraint_name = 'device_running_thresholds_device_id_fkey'
            AND table_name = 'device_running_thresholds'
            AND table_schema = 'public'
    ) THEN
        ALTER TABLE public.device_running_thresholds
        ADD CONSTRAINT device_running_thresholds_device_id_fkey
        FOREIGN KEY (device_id)
        REFERENCES public.dim_devices(id)
        ON UPDATE CASCADE
        ON DELETE CASCADE;

        RAISE NOTICE '✅ 成功为 device_running_thresholds.device_id 添加外键约束';
    ELSE
        RAISE NOTICE '⚠️ device_running_thresholds.device_id 的外键约束已存在，跳过';
    END IF;
END $$;

### ✅ 步骤2.3: 修改sp_refresh_device_running_thresholds_current存储过程

**操作描述**: 修改文件 `scripts/sql/migrations/052_create_sp_refresh_device_running_thresholds_current.sql`

**修改位置**: 第112-133行之间,在final_thresholds CTE之后,UPDATE语句之前

**修改内容**: 插入新的CTE `final_thresholds_adaptive`

**具体修改**:
```sql
-- 在第112行之后插入:
    ),
    -- 根据运行模式调整阈值策略
    final_thresholds_adaptive AS (
        SELECT
            ft.device_id,
            ft.sample_count,
            ft.threshold AS base_threshold,
            ft.mad,
            drt.running_mode,
            -- 根据运行模式调整i_off
            CASE drt.running_mode
                WHEN 'continuous' THEN 
                    (SELECT PERCENTILE_CONT(0.10) WITHIN GROUP (ORDER BY cd.max_current) 
                     FROM current_data cd WHERE cd.device_id = ft.device_id)
                WHEN 'occasional' THEN 
                    (SELECT PERCENTILE_CONT(0.05) WITHIN GROUP (ORDER BY cd.max_current) 
                     FROM current_data cd WHERE cd.device_id = ft.device_id)
                ELSE ft.threshold - GREATEST(0.01, 2.0 * ft.mad)
            END AS i_off_adjusted,
            -- 根据运行模式调整i_on
            CASE drt.running_mode
                WHEN 'continuous' THEN 
                    (SELECT PERCENTILE_CONT(0.30) WITHIN GROUP (ORDER BY cd.max_current) 
                     FROM current_data cd WHERE cd.device_id = ft.device_id)
                WHEN 'occasional' THEN 
                    (SELECT PERCENTILE_CONT(0.50) WITHIN GROUP (ORDER BY cd.max_current) 
                     FROM current_data cd WHERE cd.device_id = ft.device_id)
                ELSE ft.threshold + GREATEST(0.01, 2.0 * ft.mad)
            END AS i_on_adjusted
        FROM device_thresholds ft
        JOIN device_running_thresholds drt ON ft.device_id = drt.device_id
    )
    -- 更新阈值表(修改原有UPDATE语句)
    UPDATE device_running_thresholds t
    SET 
        enable_i = TRUE,
        i_on = fta.i_on_adjusted,
        i_off = fta.i_off_adjusted,
        threshold_quality = CASE
            WHEN fta.sample_count >= 1000 THEN 'high'
            WHEN fta.sample_count >= 500 THEN 'medium'
            WHEN fta.sample_count >= 50 THEN 'low'
            ELSE 'insufficient'
        END,
        updated_at = NOW(),
        updated_by = 'sp_refresh_device_running_thresholds_current_v2'
    FROM final_thresholds_adaptive fta
    WHERE t.device_id = fta.device_id;
```

      IF v_use_weighted THEN
        -- 加权融合逻辑
        v_score_i := CASE WHEN v_ei AND v_i_on > v_i_off 
                          THEN GREATEST(0, LEAST(1, (r_max_i - v_i_off) / NULLIF(v_i_on - v_i_off, 0)))
                          ELSE 0 END;
        v_score_p := CASE WHEN v_ep AND v_p_on > v_p_off 
                          THEN GREATEST(0, LEAST(1, (r_p - v_p_off) / NULLIF(v_p_on - v_p_off, 0)))
                          ELSE 0 END;
        v_score_f := CASE WHEN v_ef AND v_f_on > v_f_off 
                          THEN GREATEST(0, LEAST(1, (r_f - v_f_off) / NULLIF(v_f_on - v_f_off, 0)))
                          ELSE 0 END;
        v_composite := v_w_i * v_score_i + v_w_p * v_score_p + v_w_f * v_score_f;

      ts_bucket := r_ts; is_running := v_state; max_i := r_max_i; p := r_p; f := r_f;

      RETURN NEXT;
      CONTINUE;

-- 修改mark CTE(第130-149行):
  ), mark AS (
    SELECT 
      s2.ts_bucket,
      s2.is_running,
      s2.max_i,  -- 需要从fn_running_state_1s返回
      CASE
        WHEN s2.is_running THEN 1
        WHEN EXISTS (
          SELECT 1 FROM edges e
          WHERE e.kind='start'
            AND s2.ts_bucket >= e.ts
            AND s2.ts_bucket < e.ts + (v_startup_secs || ' seconds')::interval
        ) THEN 2
        WHEN EXISTS (
          SELECT 1 FROM edges e
          WHERE e.kind='stop'
            AND s2.ts_bucket >= e.ts
            AND s2.ts_bucket < e.ts + (v_shutdown_secs || ' seconds')::interval
        ) THEN 3
        ELSE 0
      END AS phase,
      -- 识别启动电流冲击
      CASE
        WHEN s2.is_running 
         AND s2.max_i > v_i_on * v_inrush_mult
         AND EXISTS (
           SELECT 1 FROM edges e
           WHERE e.kind='start'
             AND s2.ts_bucket >= e.ts
             AND s2.ts_bucket < e.ts + (v_inrush_dur || ' seconds')::interval
         )
        THEN 'startup_inrush'
        ELSE NULL
      END AS phase_type
    FROM s2
  )
