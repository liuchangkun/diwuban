```python
from functools import lru_cache
from typing import Any, Optional
import threading

class CacheManager:
    """缓存管理器（单例模式）"""
    
    _instance = None
    _lock = threading.Lock()
    
    def __new__(cls):
        """单例模式"""
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
        return cls._instance
    
    def __init__(self):
        """初始化"""
        if not hasattr(self, '_initialized'):
            self._cache = {}
            self._initialized = True
    
    def get(self, key: str) -> Optional[Any]:
        """
        获取缓存
        
        Args:
            key: 缓存键
            
        Returns:
            value: 缓存值（None表示未命中）
        """
        return self._cache.get(key)
    
    def set(self, key: str, value: Any, ttl: int = 3600) -> None:
        """
        设置缓存
        
        Args:
            key: 缓存键
            value: 缓存值
            ttl: 过期时间（秒），默认1小时
        """
        self._cache[key] = value
        # TODO: 实现TTL过期机制
    
    def clear(self) -> None:
        """清除所有缓存"""
        self._cache.clear()
    
    def clear_pattern(self, pattern: str) -> None:
        """
        清除匹配模式的缓存
        
        Args:
            pattern: 模式字符串（支持通配符*）
        """
        import fnmatch
        keys_to_delete = [k for k in self._cache.keys() if fnmatch.fnmatch(k, pattern)]
        for key in keys_to_delete:
            del self._cache[key]
```

```python
import pandas as pd
from typing import List, Callable, Any
import logging

logger = logging.getLogger(__name__)

class BatchProcessor:
    """批处理器"""
    
    def __init__(self, batch_size: int = 25000):
        """
        初始化
        
        Args:
            batch_size: 批大小，默认25000
        """
        self.batch_size = batch_size
    
    def split_data(self, data: pd.DataFrame) -> List[pd.DataFrame]:
        """
        将数据分批
        
        Args:
            data: 原始数据
            
        Returns:
            batches: 数据批次列表
        """
        n_rows = len(data)
        if n_rows <= self.batch_size:
            return [data]
        
        n_batches = (n_rows + self.batch_size - 1) // self.batch_size
        batches = []
        for i in range(n_batches):
            start_idx = i * self.batch_size
            end_idx = min((i + 1) * self.batch_size, n_rows)
            batches.append(data.iloc[start_idx:end_idx])
        
        logger.info(f"数据分为{n_batches}批，每批最多{self.batch_size}行")
        return batches
    
    def process_batches(
        self, 
        data: pd.DataFrame, 
        process_func: Callable[[pd.DataFrame], Any]
    ) -> Any:
        """
        批量处理数据
        
        Args:
            data: 原始数据
            process_func: 处理函数
            
        Returns:
            result: 处理结果
        """
        if len(data) <= self.batch_size:
            return process_func(data)
        
        batches = self.split_data(data)
        results = []
        for i, batch in enumerate(batches):
            logger.info(f"处理第{i+1}/{len(batches)}批")
            result = process_func(batch)
            results.append(result)
        
        return self.merge_results(results)
    
    def merge_results(self, results: List[Any]) -> Any:
        """
        合并批处理结果
        
        Args:
            results: 结果列表
            
        Returns:
            merged_result: 合并后的结果
        """
        # 根据结果类型合并
        if isinstance(results[0], pd.DataFrame):
            return pd.concat(results, ignore_index=True)
        elif isinstance(results[0], dict):
            # 合并字典（简单合并，可能需要更复杂的逻辑）
            merged = {}
            for result in results:
                merged.update(result)
            return merged
        else:
            return results
```

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Dict, Any
import numpy as np

@dataclass
class MethodResult:
    """方法拟合结果"""
    fit_params: Dict[str, float]        # 拟合参数
    r_squared: float                    # R²
    rmse: float                         # 均方根误差
    mae: float                          # 平均绝对误差
    mape: float                         # 平均绝对百分比误差
    metadata: Dict[str, Any]            # 其他元数据

class BaseMethod(ABC):
    """拟合方法基类"""
    
    def __init__(self, **kwargs):
        """
        初始化
        
        Args:
            **kwargs: 方法参数
        """
        self.params = kwargs
    
    @abstractmethod
    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """
        拟合方法
        
        Args:
            X: 自变量数组（流量Q）
            y: 因变量数组（扬程H/功率P/效率η）
            **kwargs: 其他参数
            
        Returns:
            result: 拟合结果
            
        Raises:
            ValueError: 如果数据无效
            FitError: 如果拟合失败
        """
        pass
    
    @abstractmethod
    def predict(self, X: np.ndarray, params: Dict[str, float]) -> np.ndarray:
        """
        预测方法
        
        Args:
            X: 自变量数组
            params: 拟合参数
            
        Returns:
            y_pred: 预测值数组
        """
        pass
    
    def get_parameters(self) -> Dict[str, Any]:
        """获取方法参数"""
        return self.params
    
    def _calculate_metrics(
        self, 
        y_true: np.ndarray, 
        y_pred: np.ndarray
    ) -> Dict[str, float]:
        """
        计算评估指标
        
        Args:
            y_true: 真实值
            y_pred: 预测值
            
        Returns:
            metrics: 评估指标字典
        """
        # R²
        ss_res = np.sum((y_true - y_pred) ** 2)
        ss_tot = np.sum((y_true - np.mean(y_true)) ** 2)
        r_squared = 1 - (ss_res / ss_tot)
        
        # RMSE
        rmse = np.sqrt(np.mean((y_true - y_pred) ** 2))
        
        # MAE
        mae = np.mean(np.abs(y_true - y_pred))
        
        # MAPE
        mape = np.mean(np.abs((y_true - y_pred) / y_true)) * 100
        
        return {
            'r_squared': r_squared,
            'rmse': rmse,
            'mae': mae,
            'mape': mape
        }
```
