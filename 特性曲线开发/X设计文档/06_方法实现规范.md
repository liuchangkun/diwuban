## 1. 方法接口规范

### 1.1 基础接口

所有拟合方法必须实现以下接口：

```python
class FittingMethod:
    """拟合方法基类"""
    
    def __init__(self, method_name: str, method_config: dict):
        """
        初始化方法
        
        Parameters:
        -----------
        method_name : str
            方法名称，如 'math_quadratic_decay'
        method_config : dict
            方法配置参数
        """
        self.method_name = method_name
        self.config = method_config
        self.params = None
        self.metadata = {}
    
    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> dict:
        """
        拟合方法
        
        Parameters:
        -----------
        X : np.ndarray
            自变量数据，shape=(n_samples, n_features)
            - 单变量：shape=(n_samples, 1)，如Q
            - 多变量：shape=(n_samples, n_features)，如(Q, f)
        y : np.ndarray
            因变量数据，shape=(n_samples,)
        **kwargs : dict
            其他参数
        
        Returns:
        --------
        result : dict
            拟合结果，包含：
            - 'params': 拟合参数
            - 'r2': R²值
            - 'rmse': RMSE值
            - 'mae': MAE值
            - 'metadata': 元数据（如迭代次数、收敛状态等）
        """
        raise NotImplementedError
    
    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        预测方法
        
        Parameters:
        -----------
        X : np.ndarray
            自变量数据，shape=(n_samples, n_features)
        
        Returns:
        --------
        y_pred : np.ndarray
            预测值，shape=(n_samples,)
        """
        raise NotImplementedError
    
    def validate_constraints(self, X: np.ndarray, y_pred: np.ndarray) -> dict:
        """
        验证物理约束
        
        Parameters:
        -----------
        X : np.ndarray
            自变量数据
        y_pred : np.ndarray
            预测值
        
        Returns:
        --------
        validation_result : dict
            验证结果，包含：
            - 'is_valid': bool，是否满足所有约束
            - 'violations': list，违反的约束列表
            - 'violation_details': dict，违反详情
        """
        raise NotImplementedError
    
    def get_metadata(self) -> dict:
        """
        获取方法元数据
        
        Returns:
        --------
        metadata : dict
            元数据，包含：
            - 'method_name': 方法名称
            - 'method_type': 方法类型（math/physics/ml/hybrid/timeseries/ensemble）
            - 'applicable_curves': 适用曲线类型列表
            - 'expected_r2': 预期R²范围
            - 'data_requirements': 数据要求
            - 'computational_cost': 计算成本（low/medium/high）
        """
        return self.metadata
```

### 1.2 方法注册机制

所有方法必须通过注册机制注册到系统中：

```python
# 方法注册器
class MethodRegistry:
    """方法注册器"""
    
    _methods = {}
    
    @classmethod
    def register(cls, method_name: str, method_class: type):
        """注册方法"""
        cls._methods[method_name] = method_class
    
    @classmethod
    def get_method(cls, method_name: str) -> type:
        """获取方法类"""
        return cls._methods.get(method_name)
    
    @classmethod
    def list_methods(cls) -> list:
        """列出所有方法"""
        return list(cls._methods.keys())

# 使用装饰器注册方法
@MethodRegistry.register('math_quadratic_decay', MathQuadraticDecay)
class MathQuadraticDecay(FittingMethod):
    """二次衰减方法"""
    pass
```

### 2.2 参数自动估计

对于需要初始猜测的方法，提供自动估计功能：

```python
def estimate_initial_params(X: np.ndarray, y: np.ndarray, method_name: str) -> dict:
    """
    自动估计初始参数
    
    Parameters:
    -----------
    X : np.ndarray
        自变量数据
    y : np.ndarray
        因变量数据
    method_name : str
        方法名称
    
    Returns:
    --------
    initial_params : dict
        初始参数估计
    """
    if method_name == 'math_quadratic_decay':
        # H = H0 - a×Q - b×Q²
        H0 = y[0]  # 假设第一个点是Q=0
        a = (y[0] - y[-1]) / X[-1, 0]  # 粗略估计斜率
        b = 0.001  # 小的二次项系数
        return {'H0': H0, 'a': a, 'b': b}
    
    elif method_name == 'math_stat_gaussian':
        # η = a × exp(-((Q - μ)² / (2σ²)))
        a = np.max(y)  # 峰值
        mu = X[np.argmax(y), 0]  # 峰值位置
        sigma = (X[-1, 0] - X[0, 0]) / 6  # 粗略估计宽度
        return {'a': a, 'mu': mu, 'sigma': sigma}
    
    # ... 其他方法
```

## 3. 物理约束集成规范

### 3.1 约束类型

物理约束分为三类：

1. **参数约束**：对拟合参数的约束
   - 示例：H0 > 0, a > 0, b > 0

2. **预测约束**：对预测结果的约束
   - 示例：H > 0, 0 ≤ η ≤ 1, dH/dQ < 0

3. **关系约束**：多个变量之间的关系
   - 示例：P = ρgQH/η, Q ∝ f, H ∝ f²

---

### 3.2 约束集成模式

**模式1：后验验证**（Post-hoc Validation）
- 先拟合，后验证
- 如果违反约束，发出警告或重新拟合
- 适用于所有方法
- 100%自动化

```python
def fit_with_posthoc_validation(method, X, y):
    # 1. 拟合
    result = method.fit(X, y)
    
    # 2. 预测
    y_pred = method.predict(X)
    
    # 3. 验证约束
    validation = method.validate_constraints(X, y_pred)
    
    # 4. 如果违反约束，处理
    if not validation['is_valid']:
        # 选项A：发出警告
        warnings.warn(f"Constraints violated: {validation['violations']}")
        
        # 选项B：重新拟合（调整参数或方法）
        # result = method.fit(X, y, enforce_constraints=True)
    
    return result
```

**模式2：约束优化**（Constrained Optimization）
- 在优化过程中强制约束
- 使用约束优化算法（如SLSQP）
- 适用于数学方法和部分机器学习方法
- 100%自动化

```python
from scipy.optimize import minimize

def fit_with_constrained_optimization(method, X, y):
    # 定义目标函数
    def objective(params):
        y_pred = method.predict_with_params(X, params)
        return np.sum((y - y_pred)**2)
    
    # 定义约束
    constraints = [
        {'type': 'ineq', 'fun': lambda p: p[0]},  # H0 > 0
        {'type': 'ineq', 'fun': lambda p: p[1]},  # a > 0
        {'type': 'ineq', 'fun': lambda p: p[2]},  # b > 0
    ]
    
    # 优化
    result = minimize(objective, x0=initial_params, constraints=constraints)
    
    return result
```

**模式3：物理嵌入**（Physics-Embedded）
- 方法本身包含物理模型
- 自动满足物理约束
- 适用于物理模型和混合方法
- 100%自动化

```python
class PhysicsPumpChar(FittingMethod):
    """泵特性方程（物理嵌入）"""
    
    def predict(self, X):
        Q = X[:, 0]
        H0, K1, K2 = self.params
        
        # 物理模型：H = H0 - K1×Q - K2×Q²
        H = H0 - K1 * Q - K2 * Q**2
        
        # 自动满足约束（通过参数约束）
        # H0 > 0, K1 > 0, K2 > 0 → H单调递减
        
        return H
```

## 4. 测试验证规范

### 4.1 单元测试

每个方法必须通过以下单元测试：

```python
import pytest
import numpy as np

class TestMathQuadraticDecay:
    """math_quadratic_decay方法的单元测试"""
    
    def test_fit_basic(self):
        """测试基本拟合功能"""
        # 生成测试数据
        Q = np.linspace(0, 300, 100)
        H = 120 - 0.2*Q - 0.001*Q**2
        
        # 拟合
        method = MathQuadraticDecay()
        result = method.fit(Q.reshape(-1, 1), H)
        
        # 验证
        assert result['r2'] > 0.99
        assert np.abs(result['params']['H0'] - 120) < 1
        assert np.abs(result['params']['a'] - 0.2) < 0.01
        assert np.abs(result['params']['b'] - 0.001) < 0.0001
    
    def test_constraints(self):
        """测试物理约束"""
        Q = np.linspace(0, 300, 100)
        H = 120 - 0.2*Q - 0.001*Q**2
        
        method = MathQuadraticDecay()
        method.fit(Q.reshape(-1, 1), H)
        
        # 验证参数约束
        assert method.params['H0'] > 0
        assert method.params['a'] > 0
        assert method.params['b'] > 0
    
    def test_prediction(self):
        """测试预测功能"""
        Q_train = np.linspace(0, 300, 100)
        H_train = 120 - 0.2*Q_train - 0.001*Q_train**2
        
        method = MathQuadraticDecay()
        method.fit(Q_train.reshape(-1, 1), H_train)
        
        # 预测
        Q_test = np.array([50, 100, 150, 200])
        H_pred = method.predict(Q_test.reshape(-1, 1))
        H_true = 120 - 0.2*Q_test - 0.001*Q_test**2
        
        # 验证
        assert np.allclose(H_pred, H_true, rtol=0.01)
```

## 5. 性能优化规范

### 5.1 计算效率

- **快速方法**（<1秒）：线性回归、简单多项式
- **中速方法**（1-10秒）：样条、核方法、浅层神经网络
- **慢速方法**（>10秒）：深度学习、高斯过程、组合方法

### 5.2 内存优化

- 大数据集（>10,000点）使用批处理
- 避免不必要的数据复制
- 及时释放中间变量

## 6. 错误处理规范

### 6.1 异常类型

```python
class FittingError(Exception):
    """拟合错误基类"""
    pass

class ConvergenceError(FittingError):
    """收敛错误"""
    pass

class ConstraintViolationError(FittingError):
    """约束违反错误"""
    pass

class InsufficientDataError(FittingError):
    """数据不足错误"""
    pass
```

### 6.2 错误处理

```python
def fit_with_error_handling(method, X, y):
    try:
        result = method.fit(X, y)
        return result
    except ConvergenceError as e:
        # 尝试调整参数重新拟合
        result = method.fit(X, y, max_iterations=5000)
        return result
    except ConstraintViolationError as e:
        # 切换到更鲁棒的方法
        fallback_method = get_fallback_method(method.method_name)
        result = fallback_method.fit(X, y)
        return result
    except InsufficientDataError as e:
        # 使用更简单的方法
        simple_method = get_simple_method(method.method_name)
        result = simple_method.fit(X, y)
        return result
```
