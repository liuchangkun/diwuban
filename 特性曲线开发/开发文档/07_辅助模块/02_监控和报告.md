```python
def _generate_summary(self, fit_result: 'FitResult') -> str:
    """模块1: 报告摘要（P0必须实现）"""
    return f"""

"""

def _generate_parameters(self, fit_result: 'FitResult') -> str:
    """模块3: 参数估计结果（P0必须实现）"""
    params_table = "| 参数 | 值 |\n|------|------|\n"
    for key, value in fit_result.coefficients.items():
        params_table += f"| {key} | {value:.6f} |\n"
    return f"\n## 3. 参数估计结果\n\n{params_table}"

def _generate_statistics(self, fit_result: 'FitResult') -> str:
    """模块4: 统计检验指标（P0必须实现）"""
    return f"""
## 4. 统计检验指标

| 指标 | 值 | 说明 |
|------|------|------|
| R² | {fit_result.r_squared:.4f} | 决定系数 |
| RMSE | {fit_result.rmse:.4f} | 均方根误差 |
| MAE | {fit_result.mae:.4f} | 平均绝对误差 |
| MAPE | {fit_result.mape:.2f}% | 平均绝对百分比误差 |
"""

def _generate_visualization_section(self, image_paths: Dict[str, Path]) -> str:
    """模块6: 可视化图表（P0必须实现）"""
    section = "\n## 6. 可视化图表\n\n"
    for name, path in image_paths.items():
        section += f"### {name}\n\n![{name}]({path})\n\n"
    return section

def _get_quality_stars(self, r_squared: float) -> int:
    """根据R²计算质量星级"""
    if r_squared >= 0.99: return 5
    elif r_squared >= 0.95: return 4
    elif r_squared >= 0.90: return 3
    elif r_squared >= 0.80: return 2
    else: return 1
```

### 5.2 告警触发逻辑

**告警触发流程**：

```python
from app.core.error_analysis import ErrorAnalyzer
from app.core.config.loader_new import load_settings
import logging

logger = logging.getLogger(__name__)

class AlertManager:
    """告警管理器"""

    def __init__(self):
        self.settings = load_settings()
        self.error_analyzer = ErrorAnalyzer()
        self.alert_rules = self.settings.alerting.rules

    def check_alerts(self):
        """检查所有告警规则"""
        for rule in self.alert_rules:
            if self._should_trigger_alert(rule):
                self._send_alert(rule)

    def _should_trigger_alert(self, rule: dict) -> bool:
        """判断是否应该触发告警"""
        metric_value = self._get_metric_value(rule['metric'], rule['window'])

        operator = rule.get('operator', 'greater_than')
        threshold = rule['threshold']

        if operator == 'greater_than':
            return metric_value > threshold
        elif operator == 'less_than':
            return metric_value < threshold
        else:
            return False

    def _get_metric_value(self, metric: str, window: int) -> float:
        """获取指标值"""
        if metric == 'error_rate':
            trends = self.error_analyzer.analyze_trends(time_window=window)
            return trends.error_rate
        elif metric == 'fitting_duration_ms':
            # 从性能监控获取
            return self._get_avg_duration(window)
        # ... 其他指标
        return 0.0

    def _send_alert(self, rule: dict):
        """发送告警"""
        severity = rule['severity']
        message = rule['message']

        # 检查是否在抑制窗口内
        if self._is_suppressed(rule):
            return

        # 根据严重程度选择通知渠道
        channels = self.settings.alerting.severity_levels[severity].notification_channels

        for channel in channels:
            if channel == 'email':
                self._send_email_alert(rule)
            elif channel == 'sms':
                self._send_sms_alert(rule)
            elif channel == 'webhook':
                self._send_webhook_alert(rule)

        # 记录告警
        logger.error(
            f"[告警] {message}",
            extra={
                "extra_data": {
                    "规则名称": rule['name'],
                    "严重程度": severity,
                    "阈值": rule['threshold'],
                    "当前值": self._get_metric_value(rule['metric'], rule['window'])
                }
            }
        )
```

```python
        if result.r_squared < 0.85:
            alert_manager.check_alerts()

        return result
    finally:
        duration = (time.time() - start_time) * 1000

        # 检查性能告警
        if duration > 30000:
            alert_manager.check_alerts()
```

### 5.4 告警最佳实践

**告警设计原则**：
1. ✅ **可操作性**：每个告警都应该有明确的处理步骤
2. ✅ **准确性**：避免误报，设置合理的阈值
3. ✅ **及时性**：关键告警应该立即通知
4. ✅ **分级管理**：根据严重程度选择通知渠道
5. ✅ **避免告警疲劳**：使用抑制规则减少重复告警

**告警阈值调优**：
- 初始阈值：基于历史数据的95分位数
- 定期评估：每月评审告警触发情况
- 动态调整：根据业务变化调整阈值

**告警响应流程**：
1. **接收告警** → 2. **确认问题** → 3. **定位原因** → 4. **执行修复** → 5. **验证恢复** → 6. **总结复盘**
