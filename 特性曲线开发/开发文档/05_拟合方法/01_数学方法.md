## 1.3 序列化和存储原则（v2.6新增）

> **⚠️ 强制规则**：所有拟合方法必须遵守以下序列化原则

### 1.3.1 禁止存储Lambda函数

**问题**：Lambda函数无法序列化到数据库JSONB字段

**解决方案**：
- ✅ **只存储系数/参数**：将拟合结果的系数存储到`coefficients`字段
- ✅ **需要时重建函数**：从系数重建预测函数（在内存中）
- ❌ **禁止存储函数对象**：不要将Lambda函数、闭包、函数引用存储到数据库

**示例**：
```python
# ❌ 错误：存储Lambda函数（无法序列化）
predict_func = lambda x: a0 + a1 * x + a2 * x**2
result = MethodResult(
    coefficients={'a0': a0, 'a1': a1, 'a2': a2},
    predict_func=predict_func  # ❌ 无法序列化
)

# ✅ 正确：只存储系数，需要时重建函数
result = MethodResult(
    coefficients={'a0': a0, 'a1': a1, 'a2': a2}
    # predict_func 不存储，需要时从系数重建
)

# 使用时重建函数
def rebuild_predict_func(coefficients):
    a0, a1, a2 = coefficients['a0'], coefficients['a1'], coefficients['a2']
    return lambda x: a0 + a1 * x + a2 * x**2
```

### 1.3.2 可序列化的数据类型

**允许存储到JSONB的类型**：
- ✅ 基本类型：int, float, str, bool
- ✅ 容器类型：dict, list
- ✅ NumPy数值：需转换为Python基本类型（`float(value)`）
- ✅ 参数名称：字符串形式的参数标识

**禁止存储的类型**：
- ❌ 函数对象：lambda, 函数引用, 闭包
- ❌ 类实例：除非实现了序列化接口
- ❌ NumPy数组：需转换为list
- ❌ 复杂对象：scipy对象、模型对象等

### 2.2 MathPoly2Method 完整实现（2次多项式）

```python
import numpy as np
from typing import Dict, Any
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MathPoly2Method(BaseMethod):
    """2次多项式拟合方法

    数学公式:
        Q-H曲线: H = a0 + a1*Q + a2*Q²
        Q-P曲线: P = a0 + a1*Q + a2*Q²
    """

    method_id = 'math_poly_2'
    method_name = '2次多项式'
    applicable_curves = ['qh', 'qp']

    def __init__(self, regularization: str = None, lambda_: float = 0.01):
        """
        初始化

        Args:
            regularization: 正则化类型（'l1', 'l2', None）
            lambda_: 正则化系数
        """
        super().__init__(method_name='2次多项式', method_id='math_poly_2')
        self.regularization = regularization
        self.lambda_ = lambda_
        self.degree = 2

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """
        执行拟合

        Args:
            X: 自变量数组
            y: 因变量数组
            **kwargs: 额外参数

        Returns:
            MethodResult: 拟合结果
        """
        # 构建设计矩阵
        A = np.vstack([np.ones_like(X), X, X**2]).T

        # v2.7修复P1-4：实现正则化
        if self.regularization == 'ridge' or self.regularization == 'l2':
            # 岭回归（L2正则化）
            I = np.eye(3)
            I[0, 0] = 0  # 不惩罚截距
            coeffs = np.linalg.solve(A.T @ A + self.lambda_ * I, A.T @ y)
            a0, a1, a2 = coeffs
        elif self.regularization == 'lasso' or self.regularization == 'l1':
            # Lasso回归（L1正则化）- 需要使用sklearn
            from sklearn.linear_model import Lasso
            lasso = Lasso(alpha=self.lambda_, fit_intercept=False)
            lasso.fit(A, y)
            a0, a1, a2 = lasso.coef_
        else:
            # 普通最小二乘
            coefficients = np.polyfit(X, y, self.degree)
            a2, a1, a0 = coefficients

        # 计算预测值
        y_pred = a0 + a1 * X + a2 * X**2

        # 计算指标
        metrics = self._calculate_metrics(y, y_pred)

        # v2.6修复：不存储Lambda函数，只存储系数
        # 需要预测时，从系数重建函数：lambda x: a0 + a1*x + a2*x**2

        return MethodResult(
            method_id=self.method_id,
            coefficients={
                'a0': float(a0),
                'a1': float(a1),
                'a2': float(a2)
            },
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
            predict_func=predict_func,
            formula=f"y = {a0:.4f} + {a1:.4f}*x + {a2:.4f}*x²",
            metadata={'regularization': self.regularization, 'lambda': self.lambda_}
        )

    def _calculate_metrics(self, y_true: np.ndarray, y_pred: np.ndarray) -> Dict[str, float]:
        """计算拟合指标"""
        ss_res = np.sum((y_true - y_pred) ** 2)
        ss_tot = np.sum((y_true - np.mean(y_true)) ** 2)
        r_squared = 1 - (ss_res / ss_tot) if ss_tot > 0 else 0

        rmse = np.sqrt(np.mean((y_true - y_pred) ** 2))
        mae = np.mean(np.abs(y_true - y_pred))

        # 计算 MAPE（避免除零）
        non_zero_mask = np.abs(y_true) > 1e-10
        if np.any(non_zero_mask):
            mape = np.mean(np.abs((y_true[non_zero_mask] - y_pred[non_zero_mask]) / y_true[non_zero_mask]))
        else:
            mape = 0.0

        return {
            'r_squared': r_squared,
            'rmse': rmse,
            'mae': mae,
            'mape': mape
        }
```

**约束条件**：
- Q-H: a₂ < 0 (保证递减)
- Q-P: a₁ > 0, a₂ < 0 (保证递增且上凸)

```python
import numpy as np
from typing import Dict, Any
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MathPoly3Method(BaseMethod):
    """3次多项式拟合方法

    数学公式:
        y = a0 + a1*x + a2*x² + a3*x³

    适用曲线: Q-H, Q-P, Q-η
    """

    method_id = 'math_poly_3'
    method_name = '3次多项式'
    applicable_curves = ['qh', 'qp', 'qeta']

    def __init__(self, regularization: str = None, lambda_: float = 0.01):
        super().__init__(method_name='3次多项式', method_id='math_poly_3')
        self.regularization = regularization
        self.lambda_ = lambda_

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """执行拟合"""
        # 构建设计矩阵
        A = np.vstack([np.ones_like(X), X, X**2, X**3]).T

        if self.regularization == 'ridge':
            # 岭回归
            I = np.eye(4)
            I[0, 0] = 0  # 不惩罚截距
            coeffs = np.linalg.solve(A.T @ A + self.lambda_ * I, A.T @ y)
        else:
            # 普通最小二乘
            coeffs, _, _, _ = np.linalg.lstsq(A, y, rcond=None)

        a0, a1, a2, a3 = coeffs

        # 计算预测值
        y_pred = a0 + a1 * X + a2 * X**2 + a3 * X**3

        # 计算指标
        metrics = self._calculate_metrics(y, y_pred)

        # v2.6修复：不存储Lambda函数，只存储系数
        # 需要预测时，从系数重建函数：lambda x: a0 + a1*x + a2*x**2 + a3*x**3

        return MethodResult(
            method_id=self.method_id,
            coefficients={'a0': float(a0), 'a1': float(a1), 'a2': float(a2), 'a3': float(a3)},
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
            predict_func=predict_func,
            formula=f"y = {a0:.4f} + {a1:.4f}x + {a2:.6f}x² + {a3:.8f}x³",
            metadata={}
        )
```

```python
from scipy.interpolate import CubicSpline
from typing import Dict, Any
from app.services.characteristic_curves.methods.base_method import BaseMethod

class MathSplineCubicMethod(BaseMethod):
    """三次样条拟合方法

    数学原理:
        在每个区间使用三次多项式，保证整体二阶连续可微
        S_i(x) = a_i + b_i(x-x_i) + c_i(x-x_i)² + d_i(x-x_i)³

    适用曲线: 所有
    """

    method_name = '三次样条'

        初始化

        Args:

        self.bc_type = bc_type
        self._spline = None

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """执行拟合"""
        # 排序数据
        sorted_idx = np.argsort(X)
        X_sorted = X[sorted_idx]
        y_sorted = y[sorted_idx]

        # 创建三次样条
        self._spline = CubicSpline(X_sorted, y_sorted, bc_type=self.bc_type)

        # 计算预测值
        y_pred = self._spline(X)

        # 计算指标
        metrics = self._calculate_metrics(y, y_pred)

        # v2.6修复：不存储Lambda函数，只存储样条参数
        # 需要预测时，从存储的knots和coefficients重建CubicSpline对象

        return MethodResult(
            method_id=self.method_id,
            coefficients={'bc_type': self.bc_type, 'n_knots': len(X_sorted)},
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
```

### 3.3 MathSplineBsplineMethod 完整实现

```python
import numpy as np
from scipy.interpolate import BSpline, splrep
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MathSplineBsplineMethod(BaseMethod):
    """B样条拟合方法

    数学原理:
        B样条是由B样条基函数的线性组合表示
        S(x) = Σ c_i × B_i,k(x)

    优点: 局部控制性，修改一个控制点只影响局部曲线
    适用曲线: 所有
    """

    method_id = 'math_spline_bspline'
    method_name = 'B样条'
    applicable_curves = ['qh', 'qp', 'qeta']

    def __init__(self, degree: int = 3, smoothing: float = None):
        """
        初始化

        Args:
            degree: 样条阶数（默认3，三次B样条）
            smoothing: 平滑参数（None表示精确插值）
        """
        super().__init__(method_name='B样条', method_id='math_spline_bspline')
        self.degree = degree
        self.smoothing = smoothing
        self._tck = None

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """执行拟合"""
        # 排序数据
        sorted_idx = np.argsort(X)
        X_sorted = X[sorted_idx]
        y_sorted = y[sorted_idx]

        # 创建B样条
        self._tck = splrep(X_sorted, y_sorted, k=self.degree, s=self.smoothing)

        # 计算预测值
        from scipy.interpolate import splev
        y_pred = splev(X, self._tck)

        # 计算指标
        metrics = self._calculate_metrics(y, y_pred)

        # v2.6修复：不存储Lambda函数，只存储B样条参数(tck)
        # 需要预测时，从存储的tck参数重建：splev(x, tck)

        return MethodResult(
            method_id=self.method_id,
            coefficients={'degree': self.degree, 'smoothing': self.smoothing},
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
            predict_func=predict_func,
            formula=f"B-Spline (degree={self.degree})",
            metadata={}
        )
```

```python
import numpy as np
from scipy.interpolate import RBFInterpolator
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MathKernelRbfMethod(BaseMethod):
    """径向基函数(RBF)拟合方法

    数学公式:
        f(x) = Σ w_i × φ(||x - x_i||)

    其中 φ 是径向基函数，常用选择:
        - 高斯: φ(r) = exp(-r²/(2σ²))
        - 多二次: φ(r) = √(1 + (r/ε)²)
        - 逆多二次: φ(r) = 1/√(1 + (r/ε)²)

    适用曲线: 所有
    """

    method_id = 'math_kernel_rbf'
    method_name = '径向基函数'
    applicable_curves = ['qh', 'qp', 'qeta']

    def __init__(self, kernel: str = 'thin_plate_spline', epsilon: float = None):
        """
        初始化

        Args:
            kernel: 核函数类型 ('thin_plate_spline', 'multiquadric', 'gaussian')
            epsilon: 形状参数
        """
        super().__init__(method_name='径向基函数', method_id='math_kernel_rbf')
        self.kernel = kernel
        self.epsilon = epsilon
        self._rbf = None

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """执行拟合"""
        # 重塑数据
        X_2d = X.reshape(-1, 1)

        # 创建RBF插值器
        self._rbf = RBFInterpolator(X_2d, y, kernel=self.kernel, epsilon=self.epsilon)

        # 计算预测值
        y_pred = self._rbf(X_2d)

        # 计算指标
        metrics = self._calculate_metrics(y, y_pred)

        # 创建预测函数
        def predict_func(x):
            x_arr = np.atleast_1d(x).reshape(-1, 1)
            return self._rbf(x_arr)

        return MethodResult(
            method_id=self.method_id,
            coefficients={'kernel': self.kernel, 'epsilon': self.epsilon},
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
            predict_func=predict_func,
            formula=f"RBF Interpolation (kernel={self.kernel})",
            metadata={}
        )
```

| `math_rational_pade` | Padé逼近 | 所有 | 有理逼近 |

### 5.2 MathRationalPadeMethod 完整实现

```python
import numpy as np
from scipy.optimize import curve_fit
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MathRationalPadeMethod(BaseMethod):
    """Padé逼近拟合方法

    数学公式:
        f(x) = (a0 + a1*x + a2*x²) / (1 + b1*x + b2*x²)

    优点:
        - 比多项式有更好的外推性能
        - 能够表示渐近行为
        - 适合有极点或渐近线的曲线

    适用曲线: 所有
    """

    method_id = 'math_rational_pade'
    method_name = 'Padé逼近'
    applicable_curves = ['qh', 'qp', 'qeta']

    def __init__(self, num_degree: int = 2, den_degree: int = 2):
        """
        初始化

        Args:
            num_degree: 分子多项式阶数
            den_degree: 分母多项式阶数
        """
        super().__init__(method_name='Padé逼近', method_id='math_rational_pade')
        self.num_degree = num_degree
        self.den_degree = den_degree

    @staticmethod
    def _pade_22(x, a0, a1, a2, b1, b2):
        """Padé(2,2)逼近"""
        numerator = a0 + a1 * x + a2 * x**2
        denominator = 1 + b1 * x + b2 * x**2
        return numerator / denominator

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """执行拟合"""
        # 初始猜测
        p0 = [y[0], 0.01, -0.0001, 0.001, 0.00001]

        try:
            popt, pcov = curve_fit(
                self._pade_22, X, y,
                p0=p0,
                maxfev=10000
            )

            a0, a1, a2, b1, b2 = popt

            # 计算预测值
            y_pred = self._pade_22(X, *popt)

            # 计算指标
            metrics = self._calculate_metrics(y, y_pred)

            # v2.6修复：不存储Lambda函数，只存储Pade系数
            # 需要预测时，从系数重建函数：(a0+a1*x+a2*x**2)/(1+b1*x+b2*x**2)

            return MethodResult(
                method_id=self.method_id,
```

### 6.2 MathStatGaussianMethod 完整实现

```python
import numpy as np
from scipy.optimize import curve_fit
from typing import Dict, Any
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MathStatGaussianMethod(BaseMethod):
    """高斯函数拟合方法

    数学公式:
        η = A * exp(-((Q - μ)² / (2 * σ²)))

    适用曲线: Q-η（效率曲线的单峰特性）
    """

    method_id = 'math_stat_gaussian'
    method_name = '高斯函数'
    applicable_curves = ['qeta']

    def __init__(self):
        super().__init__(method_name='高斯函数', method_id='math_stat_gaussian')

    @staticmethod
    def _gaussian(x, A, mu, sigma):
        """高斯函数"""
        return A * np.exp(-((x - mu) ** 2) / (2 * sigma ** 2))

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """
        执行拟合

        Args:
            X: 流量数组
            y: 效率数组
            **kwargs: 额外参数（包含constraints）

        Returns:
            MethodResult: 拟合结果
        """
        constraints = kwargs.get('constraints', {})

        # 初始猜测
        A_init = y.max()
        mu_init = X[np.argmax(y)]
        sigma_init = (X.max() - X.min()) / 4

        # 参数边界
        bounds = (
            [0, X.min(), 0],           # 下界
            [1.5, X.max(), X.max()]    # 上界
        )

        try:
            popt, pcov = curve_fit(
                self._gaussian, X, y,
                p0=[A_init, mu_init, sigma_init],
                bounds=bounds,
                maxfev=5000
            )

            A, mu, sigma = popt

            # 计算预测值
            y_pred = self._gaussian(X, A, mu, sigma)

            # 计算指标
            metrics = self._calculate_metrics(y, y_pred)

            # v2.6修复：不存储Lambda函数，只存储高斯参数
            # 需要预测时，从参数重建函数：A * exp(-((x-mu)**2)/(2*sigma**2))

            return MethodResult(
                method_id=self.method_id,
                coefficients={'A': float(A), 'mu': float(mu), 'sigma': float(sigma)},
                r_squared=metrics['r_squared'],
                rmse=metrics['rmse'],
                mae=metrics['mae'],
                mape=metrics['mape'],
                predict_func=predict_func,
                formula=f"η = {A:.4f} * exp(-((Q - {mu:.2f})² / (2 * {sigma:.2f}²)))",
                metadata={}
            )

        except Exception as e:
            self._logger.warning(f"高斯拟合失败: {e}")
            return MethodResult(
                method_id=self.method_id,
                coefficients={},
                r_squared=0.0,
                rmse=float('inf'),
                mae=float('inf'),
                mape=0.0,
                predict_func=None,
                formula="拟合失败",
                metadata={'error': str(e)}
            )
```

### 7.2 MathLocalLowessMethod 完整实现

```python
import numpy as np
from statsmodels.nonparametric.smoothers_lowess import lowess
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MathLocalLowessMethod(BaseMethod):
    """局部加权回归(LOWESS)拟合方法

    算法原理:
        LOWESS (Locally Weighted Scatterplot Smoothing)
        在每个点使用加权最小二乘进行局部拟合
        权重由三次权重函数确定: w(u) = (1 - |u|³)³

    优点:
        - 非参数方法，不需要假设函数形式
        - 对异常值鲁棒
        - 自动适应数据的局部特征

    适用曲线: 所有
    """

    method_id = 'math_local_lowess'
    method_name = '局部加权回归'
    applicable_curves = ['qh', 'qp', 'qeta']

    def __init__(self, frac: float = 0.3, it: int = 3):
        """
        初始化

        Args:
            frac: 用于估计每个点的数据比例 (0, 1]
            it: 鲁棒迭代次数
        """
        super().__init__(method_name='局部加权回归', method_id='math_local_lowess')
        self.frac = frac
        self.it = it
        self._smoothed = None

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """执行拟合"""
        # 执行LOWESS平滑
        self._smoothed = lowess(y, X, frac=self.frac, it=self.it, return_sorted=True)

        # 提取平滑后的值
        X_smooth = self._smoothed[:, 0]
        y_smooth = self._smoothed[:, 1]

        # 使用插值创建预测函数
        from scipy.interpolate import interp1d
        self._interp = interp1d(X_smooth, y_smooth, kind='linear', fill_value='extrapolate')

        # 计算预测值
        y_pred = self._interp(X)

        # 计算指标
        metrics = self._calculate_metrics(y, y_pred)

        # v2.6修复：不存储Lambda函数，只存储LOWESS参数
        # 需要预测时，从存储的平滑数据重建插值函数

        return MethodResult(
            method_id=self.method_id,
            coefficients={'frac': self.frac, 'it': self.it, 'n_points': len(X_smooth)},
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
            predict_func=predict_func,
            formula=f"LOWESS (frac={self.frac}, it={self.it})",
            metadata={}
        )
```
