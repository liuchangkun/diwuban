### 2.2 MLGradientBoostMethod 完整实现

```python
import numpy as np
from typing import Dict, Any
from sklearn.ensemble import GradientBoostingRegressor
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MLGradientBoostMethod(BaseMethod):
    """梯度提升拟合方法

    适用场景: 大数据量（>1000点）、复杂非线性关系

    适用曲线: 所有
    """

    method_id = 'ml_gradient_boost'
    method_name = '梯度提升'
    applicable_curves = ['qh', 'qp', 'qeta']
    min_data_points = 1000

    def __init__(
        self,
        n_estimators: int = 100,
        max_depth: int = 3,
        learning_rate: float = 0.1
    ):
        """
        初始化

        Args:
            n_estimators: 树的数量
            max_depth: 树的最大深度
            learning_rate: 学习率
        """
        super().__init__(method_name='梯度提升', method_id='ml_gradient_boost')
        self.n_estimators = n_estimators
        self.max_depth = max_depth
        self.learning_rate = learning_rate
        self._model = None

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """
        执行拟合

        Args:
            X: 自变量数组
            y: 因变量数组
            **kwargs: 额外参数

        Returns:
            MethodResult: 拟合结果
        """
        # 检查数据量
        if len(X) < self.min_data_points:
            self._logger.warning(f"数据量不足: {len(X)} < {self.min_data_points}")

        # 重塑数据
        X_reshaped = X.reshape(-1, 1)

        # 创建并训练模型
        self._model = GradientBoostingRegressor(
            n_estimators=self.n_estimators,
            max_depth=self.max_depth,
            learning_rate=self.learning_rate,
            random_state=42
        )
        self._model.fit(X_reshaped, y)

        # 计算预测值
        y_pred = self._model.predict(X_reshaped)

        # 计算指标
        metrics = self._calculate_metrics(y, y_pred)

        # 创建预测函数
        def predict_func(x):
            x_arr = np.atleast_1d(x).reshape(-1, 1)
            return self._model.predict(x_arr)

        return MethodResult(
            method_id=self.method_id,
            coefficients={'n_estimators': self.n_estimators, 'max_depth': self.max_depth},
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
            predict_func=predict_func,
            formula="Gradient Boosting Model",
            metadata={'model_type': 'GradientBoostingRegressor'}
        )
```

### 4.1 MLRandomForestMethod 完整实现

```python
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MLRandomForestMethod(BaseMethod):
    """随机森林拟合方法

    算法原理:
        集成多个决策树（Bagging），每棵树使用随机子集训练
        最终预测值为所有树预测的平均值

    优点:
        - 鲁棒性强，对噪声和异常值不敏感
        - 不易过拟合
        - 可以评估特征重要性

    适用曲线: 所有
    """

    method_id = 'ml_random_forest'
    method_name = '随机森林'
    applicable_curves = ['qh', 'qp', 'qeta']

    def __init__(
        self,
        n_estimators: int = 100,
        max_depth: int = 10,
        min_samples_split: int = 5,
        min_samples_leaf: int = 2
    ):
        """
        初始化

        Args:
            n_estimators: 树的数量
            max_depth: 树的最大深度
            min_samples_split: 分裂所需最小样本数
            min_samples_leaf: 叶节点最小样本数
        """
        super().__init__(method_name='随机森林', method_id='ml_random_forest')
        self.n_estimators = n_estimators
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.min_samples_leaf = min_samples_leaf
        self._model = None

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """执行拟合"""
        X_reshaped = X.reshape(-1, 1)

        self._model = RandomForestRegressor(
            n_estimators=self.n_estimators,
            max_depth=self.max_depth,
            min_samples_split=self.min_samples_split,
            min_samples_leaf=self.min_samples_leaf,
            random_state=42,
            n_jobs=-1
        )
        self._model.fit(X_reshaped, y)

        y_pred = self._model.predict(X_reshaped)
        metrics = self._calculate_metrics(y, y_pred)

        def predict_func(x):
            x_arr = np.atleast_1d(x).reshape(-1, 1)
            return self._model.predict(x_arr)

        return MethodResult(
            method_id=self.method_id,
            coefficients={
                'n_estimators': self.n_estimators,
                'max_depth': self.max_depth,
                'feature_importance': self._model.feature_importances_.tolist()
            },
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
            predict_func=predict_func,
            formula="Random Forest Regressor",
            metadata={}
        )
```

### 4.2 MLGaussianProcessMethod 完整实现

```python
import numpy as np
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel, WhiteKernel
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class MLGaussianProcessMethod(BaseMethod):
    """高斯过程回归拟合方法

    算法原理:
        高斯过程是一种概率模型，提供预测值的同时给出不确定性估计
        f(x) ~ GP(m(x), k(x, x'))

    优点:
        - 提供不确定性量化（置信区间）
        - 自动平滑
        - 适合中小规模数据

    适用曲线: 所有
    """

    method_id = 'ml_gaussian_process'
    method_name = '高斯过程回归'
    applicable_curves = ['qh', 'qp', 'qeta']

    def __init__(self, length_scale: float = 1.0, noise_level: float = 0.1):
        """
        初始化

        Args:
            length_scale: RBF核的长度尺度
            noise_level: 噪声水平
        """
        super().__init__(method_name='高斯过程回归', method_id='ml_gaussian_process')
        self.length_scale = length_scale
        self.noise_level = noise_level
        self._model = None

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """执行拟合"""
        X_reshaped = X.reshape(-1, 1)

        # 定义核函数
        kernel = ConstantKernel(1.0) * RBF(length_scale=self.length_scale) + WhiteKernel(noise_level=self.noise_level)

        self._model = GaussianProcessRegressor(
            kernel=kernel,
            n_restarts_optimizer=10,
            random_state=42
        )
        self._model.fit(X_reshaped, y)

        y_pred, y_std = self._model.predict(X_reshaped, return_std=True)
        metrics = self._calculate_metrics(y, y_pred)

        def predict_func(x, return_std=False):
            x_arr = np.atleast_1d(x).reshape(-1, 1)
            if return_std:
                return self._model.predict(x_arr, return_std=True)
            return self._model.predict(x_arr)

        return MethodResult(
            method_id=self.method_id,
            coefficients={
                'kernel_params': str(self._model.kernel_),
                'log_marginal_likelihood': self._model.log_marginal_likelihood_value_
            },
            r_squared=metrics['r_squared'],
            rmse=metrics['rmse'],
            mae=metrics['mae'],
            mape=metrics['mape'],
            predict_func=predict_func,
            formula="Gaussian Process Regression",
            metadata={'mean_std': float(np.mean(y_std))}
        )
```

## 6. 使用建议

### 6.1 数据量与方法选择

| 数据量 | 推荐方法 | 说明 |
|--------|---------|------|
| < 100 | 不推荐ML | 数据量不足 |
| 100-500 | `ml_svr` | 小样本学习 |
| 500-5000 | `ml_gradient_boost` | 中等规模 |
| > 5000 | `ml_mlp` | 大规模数据 |

### 6.2 注意事项

1. **过拟合风险**：ML方法容易过拟合，需要交叉验证
2. **物理约束**：ML方法不保证物理约束，需要后处理验证
3. **计算资源**：神经网络需要较多计算资源
4. **可解释性**：ML方法可解释性较差，系数无物理意义
