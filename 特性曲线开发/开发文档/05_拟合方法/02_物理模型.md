## 2. 泵特性方程（10种）

### 2.1 P0 方法

| 方法ID | 方法名称 | 适用曲线 | 说明 |
|--------|---------|---------|------|
| `physics_pump_char` | 泵特性方程 | Q-H | **推荐** |

### 2.2 PhysicsPumpCharMethod 完整实现

```python
import numpy as np
from scipy.optimize import curve_fit
from typing import Dict, Any
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class PhysicsPumpCharMethod(BaseMethod):
    """泵特性方程拟合方法

    物理公式:
        H = H0 - K * Q²

    物理意义:
        H0: 零流量扬程（截止扬程）
        K: 阻力系数

    适用曲线: Q-H
    """

    method_id = 'physics_pump_char'
    method_name = '泵特性方程'
    applicable_curves = ['qh']

    def __init__(self):
        super().__init__(method_name='泵特性方程', method_id='physics_pump_char')

    @staticmethod
    def _pump_curve(Q, H0, K):
        """泵特性方程: H = H0 - K*Q²"""
        return H0 - K * Q ** 2

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """
        执行拟合

        Args:
            X: 流量数组（归一化后）
            y: 扬程数组（归一化后）
            **kwargs: 额外参数（包含constraints）

        Returns:
            MethodResult: 拟合结果

        Raises:
            ValueError: 当输入数据不满足要求时
        """
        # v2.7修复P1-6：添加输入数据验证
        if len(X) == 0 or len(y) == 0:
            raise ValueError("输入数据不能为空")

        if len(X) != len(y):
            raise ValueError(f"X和y长度不一致: {len(X)} != {len(y)}")

        if np.any(X < 0):
            raise ValueError("流量Q不能为负值")

        if np.any(y < 0):
            raise ValueError("扬程H不能为负值")

        if np.max(X) == 0:
            raise ValueError("流量Q最大值为0，无法拟合")

        constraints = kwargs.get('constraints', {})

        # 使用约束参数作为初始猜测
        if 'H0' not in constraints:
            H0_init = y.max()
            self._logger.warning(f"约束参数中缺少H0，使用数据最大值: {H0_init:.2f}")
        else:
            H0_init = constraints['H0']
            # v2.7修复P1-6：验证H0参数
            if H0_init <= 0:
                raise ValueError(f"H0必须大于0，当前值: {H0_init}")

        if 'K' not in constraints:
            K_init = (H0_init - y.min()) / (X.max() ** 2)
        else:
            K_init = constraints['K']
            # v2.7修复P1-6：验证K参数
            if K_init <= 0:
                raise ValueError(f"K必须大于0，当前值: {K_init}")

        # 参数边界
        bounds = (
            [0, 0],                    # 下界: H0 > 0, K > 0
            [H0_init * 2, K_init * 5]  # 上界
        )

        try:
            popt, pcov = curve_fit(
                self._pump_curve, X, y,
                p0=[H0_init, K_init],
                bounds=bounds,
                maxfev=5000
            )

            H0, K = popt

            # 计算预测值
            y_pred = self._pump_curve(X, H0, K)

            # 计算指标
            metrics = self._calculate_metrics(y, y_pred)

            # 验证物理约束
            physics_valid = H0 > 0 and K > 0

            # v2.6修复：不存储Lambda函数，只存储物理参数
            # 需要预测时，从参数重建函数：H0 - K * x**2

            return MethodResult(
                method_id=self.method_id,
                coefficients={'H0': float(H0), 'K': float(K)},
                r_squared=metrics['r_squared'],
                rmse=metrics['rmse'],
                mae=metrics['mae'],
                mape=metrics['mape'],
                predict_func=predict_func,
                formula=f"H = {H0:.4f} - {K:.6f} * Q²",
                metadata={'physics_valid': physics_valid}
            )

        except Exception as e:
            self._logger.warning(f"泵特性方程拟合失败: {e}")
            return MethodResult(
                method_id=self.method_id,
                coefficients={},
                r_squared=0.0,
                rmse=float('inf'),
                mae=float('inf'),
                mape=0.0,
                predict_func=None,
                formula="拟合失败",
                metadata={'error': str(e)}
            )
```

**物理约束**：
- H₀ > 0 (关阀扬程为正)
- K > 0 (阻力系数为正)
- dH/dQ = -2KQ < 0 (单调递减，对于Q>0恒成立)

## 3. 功率方程（8种）

### 3.1 P0 方法

| 方法ID | 方法名称 | 适用曲线 | 说明 |
|--------|---------|---------|------|
| `physics_power_eq` | 功率方程 | Q-P | **推荐** |

### 3.2 PhysicsPowerEqMethod 完整实现

```python
import numpy as np
from scipy.optimize import curve_fit
from typing import Dict, Any
from app.services.characteristic_curves.methods.base_method import BaseMethod
from app.services.characteristic_curves.models import MethodResult  # v2.2修复P69

class PhysicsPowerEqMethod(BaseMethod):
    """功率方程拟合方法

    物理公式:
        P = P0 + K * Q^n

    物理意义:
        P0: 空载功率
        K: 功率系数
        n: 指数（通常为1-2）

    适用曲线: Q-P

    说明（v2.7修复P1-7）:
        本方法使用经验功率方程 P = P0 + K*Q^n，不需要效率η。

        如果需要使用理论功率方程 P = ρgQH/η，有以下选项：

        1. **从数据计算η**（推荐）:
           - 如果同时有Q、H、P数据，可计算: η = ρgQH/P
           - 然后拟合η-Q曲线，再用于P-Q预测

        2. **使用额定效率**:
           - 从设备铭牌获取η_rated
           - 假设η恒定（简化，精度较低）

        3. **使用效率曲线**:
           - 先拟合Q-η曲线（使用PhysicsEfficiencyEqMethod）
           - 再计算P = ρgQH/η(Q)

        本方法选择经验公式是因为：
        - 不依赖效率数据，适用性更广
        - 拟合简单，参数少
        - 对于Q-P曲线拟合精度足够
    """

    method_id = 'physics_power_eq'
    method_name = '功率方程'
    applicable_curves = ['qp']

    def __init__(self, use_theoretical_formula: bool = False, eta: float = None):
        """
        初始化

        Args:
            use_theoretical_formula: 是否使用理论公式 P = ρgQH/η（默认False）
            eta: 效率值（仅当use_theoretical_formula=True时使用）
        """
        super().__init__(method_name='功率方程', method_id='physics_power_eq')
        self.use_theoretical_formula = use_theoretical_formula
        self.eta = eta

    @staticmethod
    def _power_curve(Q, P0, K, n):
        """功率方程: P = P0 + K*Q^n"""
        return P0 + K * np.power(Q, n)

    def fit(self, X: np.ndarray, y: np.ndarray, **kwargs) -> MethodResult:
        """
        执行拟合

        Args:
            X: 流量数组
            y: 功率数组
            **kwargs: 额外参数

        Returns:
            MethodResult: 拟合结果
        """
        constraints = kwargs.get('constraints', {})

        # 初始猜测
        if 'P0' not in constraints:
            P0_init = y.min()
        else:
            P0_init = constraints['P0']

        K_init = (y.max() - P0_init) / (X.max() ** 1.5) if X.max() > 0 else 0.001
        n_init = 1.5

        # 参数边界
        bounds = (
            [0, 0, 0.5],      # 下界
            [y.max(), 10, 3]  # 上界
        )

        try:
            popt, pcov = curve_fit(
                self._power_curve, X, y,
                p0=[P0_init, K_init, n_init],
                bounds=bounds,
                maxfev=5000
            )

            P0, K, n = popt

            # 计算预测值
            y_pred = self._power_curve(X, P0, K, n)

            # 计算指标
            metrics = self._calculate_metrics(y, y_pred)

            # 验证物理约束
            physics_valid = P0 > 0 and K > 0 and 0.5 < n < 3

            # v2.6修复：不存储Lambda函数，只存储物理参数
            # 需要预测时，从参数重建函数：P0 + K * x**n

            return MethodResult(
                method_id=self.method_id,
                coefficients={'P0': float(P0), 'K': float(K), 'n': float(n)},
                r_squared=metrics['r_squared'],
                rmse=metrics['rmse'],
                mae=metrics['mae'],
                mape=metrics['mape'],
                predict_func=predict_func,
                formula=f"P = {P0:.4f} + {K:.6f} * Q^{n:.2f}",
                metadata={'physics_valid': physics_valid}
            )

        except Exception as e:
            self._logger.warning(f"功率方程拟合失败: {e}")
            return MethodResult(
                method_id=self.method_id,
                coefficients={},
                r_squared=0.0,
                rmse=float('inf'),
                mae=float('inf'),
                mape=0.0,
                predict_func=None,
                formula="拟合失败",
                metadata={'error': str(e)}
            )
```

**物理约束**：
- P₀ > 0 (空载功率为正)
- K > 0 (功率系数为正)
- 0.5 < n < 3 (指数在合理范围)
- dP/dQ > 0 (单调递增)

## 5. 相似定律（5种）

| 方法ID | 方法名称 | 适用曲线 | 阶段 |
|--------|---------|---------|------|
| `physics_affinity_law` | 相似定律 | 所有 | P1 |
| `physics_freq_scaling` | 频率缩放 | 所有 | P1 |

### 5.1 相似定律公式

```
Q₂/Q₁ = n₂/n₁
H₂/H₁ = (n₂/n₁)²
P₂/P₁ = (n₂/n₁)³

其中：
- n: 转速（或频率）
- Q: 流量
- H: 扬程
- P: 功率
```

**归一化到50Hz**：
```python
Q_50 = Q * (50 / f)
H_50 = H * (50 / f) ** 2
P_50 = P * (50 / f) ** 3
```
