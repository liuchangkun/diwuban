## 1. 概述

### 1.1 设计目标

采用**三表分离设计**，将拟合结果拆分为：
- `curve_fit_results`（主表）：基本信息、数据信息、元数据
- `curve_fit_params`（参数表）：拟合系数和方法参数
- `curve_fit_metrics`（指标表）：评估指标（R²、RMSE、MAE、MAPE）

### 1.2 表总览

| 表名 | 说明 | 预估数据量 | 优先级 |
|-----|------|-----------|--------|
| `curve_fit_results` | 拟合结果主表 | 100,000行/年 | P0 |
| `curve_fit_params` | 拟合参数表 | 500,000行/年 | P0 |
| `curve_fit_metrics` | 拟合指标表 | 100,000行/年 | P0 |
| `curve_fit_latest` | 最新版本视图 | 3行/设备 | P0 |
| `v_curve_fit_full` | 完整拟合结果视图 | - | P0 |
| `curve_fit_history` | 历史评估记录 | 500,000行/年 | P1 |
| `learned_constraints` | 学习约束参数表 | 18行 | P1 |
| `dim_metric_config` | 指标配置表（v2.6新增） | ~50行 | P0 |
| `fact_measurements` | 测量数据表（外部表） | 大量 | P0 |

**说明**：
- `dim_metric_config`和`fact_measurements`是外部已存在的表，本文档仅说明其结构和使用方式
- 特性曲线系统通过`metric_key`查询这两个表获取测量数据

### 2.2 外部表：fact_measurements（v2.6新增说明）

> **⚠️ 注意**：此表为外部已存在的表，本文档仅说明其结构和使用方式

```sql
-- 测量数据表（简化结构）
CREATE TABLE fact_measurements (
    id BIGSERIAL PRIMARY KEY,
    device_id INTEGER NOT NULL,               -- 设备ID
    metric_id INTEGER NOT NULL,               -- 指标ID（关联dim_metric_config.id）
    ts_bucket TIMESTAMP WITH TIME ZONE NOT NULL,  -- 时间桶
    value DOUBLE PRECISION,                   -- 测量值
    quality_code INTEGER,                     -- 质量码
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),

    CONSTRAINT fk_measurements_metric FOREIGN KEY (metric_id)
        REFERENCES dim_metric_config(id) ON DELETE RESTRICT
);

-- 索引
CREATE INDEX idx_measurements_device_metric_time
    ON fact_measurements(device_id, metric_id, ts_bucket);
```

**查询模式**：
```sql
-- 通过metric_key查询数据
SELECT ts_bucket, AVG(value) as metric_value
FROM fact_measurements
WHERE device_id = %(device_id)s
  AND metric_id = (
      SELECT id FROM dim_metric_config
      WHERE metric_key = 'pump_flow_rate'
  )
  AND ts_bucket BETWEEN %(start_time)s AND %(end_time)s
GROUP BY ts_bucket
ORDER BY ts_bucket;
```

### 2.3 主表：curve_fit_results

```sql
CREATE TABLE curve_fit_results (
    -- ========== 主键 ==========
    id SERIAL NOT NULL,

    -- ========== 基本信息 ==========
    device_id INTEGER,  -- 允许NULL（支持泵组曲线）
    curve_type VARCHAR(50) NOT NULL,
    version VARCHAR(50) NOT NULL,
    method_name VARCHAR(100) NOT NULL,

    -- ========== 数据信息 ==========
    data_start_time TIMESTAMP WITH TIME ZONE,
    data_end_time TIMESTAMP WITH TIME ZONE,
    data_point_count INTEGER,
    data_coverage_rate DOUBLE PRECISION,

    -- ========== 复杂结构（JSONB） ==========
    -- 注：JSONB字段索引策略详见 02_索引和JSONB.md 第1.5节（v3.1新增）
    -- 注：JSONB字段schema验证详见 02_索引和JSONB.md 第4节（v3.2新增）
    physics_validation JSONB,
    historical_evaluation JSONB,
    normalization_params JSONB,
    curve_points JSONB,

    -- ========== 元数据 ==========
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    created_by VARCHAR(100),
    status VARCHAR(20) DEFAULT 'active',
    tags TEXT[],
    notes TEXT,

    -- ========== 约束 ==========
    CONSTRAINT pk_curve_fit_results PRIMARY KEY (id),
    CONSTRAINT fk_results_device FOREIGN KEY (device_id)
        REFERENCES public.dim_devices(id) ON DELETE RESTRICT,
    CONSTRAINT uq_results_device_curve_version
        UNIQUE(device_id, curve_type, version),
    CONSTRAINT chk_results_curve_type
        CHECK (curve_type IN (
            -- 单泵曲线类型
            'qh', 'qp', 'qeta', 'heta', 'peta', 'qnpsh',
            -- 并联泵组曲线类型
            'parallel_qh', 'parallel_qp', 'parallel_qeta',
            -- 台数优化曲线类型
            'n_q', 'n_p', 'n_eta',
            -- 分配曲线类型
            'load_distribution', 'flow_distribution', 'power_distribution',
            -- 控制策略曲线类型
            'start_stop_strategy', 'switching_timing', 'switching_process',
            -- 其他泵组曲线类型
            'optimal_region', 'specific_energy', 'coordination_control'
        )),
    CONSTRAINT chk_results_status
        CHECK (status IN ('active', 'archived', 'deprecated'))
);
```

**字段说明**：

| 字段名 | 类型 | 约束 | 说明 |

| `device_id` | INTEGER | NULL, FK | 设备ID（单泵必填，泵组为空） |
| `curve_type` | VARCHAR(50) | NOT NULL, CHECK | 曲线类型（qh/qp/qeta等） |
| `version` | VARCHAR(50) | NOT NULL, UNIQUE | 版本号（YYYYMMDD_HHMMSS） |
| `method_name` | VARCHAR(100) | NOT NULL | 拟合方法名称 |
| `data_point_count` | INTEGER | NULL | 数据点数 |
| `physics_validation` | JSONB | NULL | 物理约束验证结果 |
| `curve_points` | JSONB | NULL | 曲线特征点数据 |
| `status` | VARCHAR(20) | DEFAULT 'active' | 状态 |

| `method` | 方法参数 | degree, lambda, n_estimators | P0 |
| `correction` | 修正系数 | alpha_intercept, alpha_coef_N | P2 |

```sql
CREATE TABLE curve_fit_metrics (

    id SERIAL NOT NULL,

    -- ========== 外键关联（1:1关系） ==========
    result_id INTEGER NOT NULL,

    -- ========== 评估指标 ==========
    r_squared DOUBLE PRECISION,
    rmse DOUBLE PRECISION,
    mae DOUBLE PRECISION,
    mape DOUBLE PRECISION,

    -- ========== 元数据 ==========
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),

    -- ========== 约束 ==========
    CONSTRAINT pk_curve_fit_metrics PRIMARY KEY (id),

        FOREIGN KEY (result_id) REFERENCES curve_fit_results(id) ON DELETE CASCADE,
    CONSTRAINT uq_metrics_result_id UNIQUE(result_id),
    CONSTRAINT chk_metrics_r_squared
        CHECK (r_squared IS NULL OR (r_squared >= 0 AND r_squared <= 1))
);
```

### 2.4 P2阶段扩展（泵组曲线支持）

> **目的**：支持泵组曲线存储，允许按泵站+泵组合查询
> **来源**：`第一次开发/06_数据库和配置设计.md` 2.1.3节

#### 2.4.1 扩展字段DDL

```sql
-- ========== P2阶段扩展：泵组曲线支持 ==========

-- 1. device_id 允许NULL（v2.5修复）
-- 说明：泵组曲线不关联单个设备
ALTER TABLE curve_fit_results ALTER COLUMN device_id DROP NOT NULL;

-- 2. 添加P2扩展字段
ALTER TABLE curve_fit_results
    ADD COLUMN IF NOT EXISTS station_id BIGINT,           -- 泵站ID
    ADD COLUMN IF NOT EXISTS pump_combination BIGINT[],   -- 泵组合（设备ID数组）
    ADD COLUMN IF NOT EXISTS pump_count INTEGER,          -- 泵台数
    ADD COLUMN IF NOT EXISTS group_type VARCHAR(30),      -- 泵组类型
    ADD COLUMN IF NOT EXISTS base_pump_id BIGINT,         -- 基准泵ID（同构泵组）
    ADD COLUMN IF NOT EXISTS correction_model JSONB;      -- 系统修正系数模型

-- 添加泵站外键
ALTER TABLE curve_fit_results
    ADD CONSTRAINT fk_results_station
        FOREIGN KEY (station_id) REFERENCES public.dim_stations(id) ON DELETE RESTRICT;
```

#### 2.4.2 P2扩展约束

```sql
-- ========== P2阶段约束 ==========

-- 约束1：泵组类型枚举
-- 注意：与 GroupProcessingStrategy 枚举保持一致（见 03_核心模块/06_数据结构.md）
ALTER TABLE curve_fit_results
    ADD CONSTRAINT chk_results_group_type
        CHECK (group_type IS NULL OR group_type IN (
            'homogeneous',          -- 同构泵组（相同型号变频泵并联）
            'heterogeneous',        -- 异构泵组（不同型号，功率差异≥10%）
            'vfd_hetero_freq',      -- 全变频异构（不同型号，频率差异≥2%）
            'mixed',                -- 混合泵组（变频泵+软启泵，功率差异<10%）
            'mixed_heterogeneous'   -- 异构混合泵组（变频泵+软启泵，功率差异≥10%）
        ));

-- 约束2：泵组曲线版本唯一性
-- 说明：同一泵站+泵组合+曲线类型只能有一个活跃版本
ALTER TABLE curve_fit_results
    ADD CONSTRAINT uq_results_group_curve_version
        UNIQUE(station_id, pump_combination, curve_type, version);

-- 约束3：单泵/泵组互斥
-- 说明：device_id 和 pump_combination 不能同时为空或同时有值
ALTER TABLE curve_fit_results
    ADD CONSTRAINT chk_results_pump_or_group
        CHECK (
            (device_id IS NOT NULL AND pump_combination IS NULL) OR
            (device_id IS NULL AND pump_combination IS NOT NULL)
        );
```

### 4.2 完整拟合结果视图：v_curve_fit_full

```sql
CREATE OR REPLACE VIEW v_curve_fit_full AS
SELECT
    r.*,
    m.r_squared,
    m.rmse,
    m.mae,
    m.mape,
    (
        SELECT jsonb_object_agg(p.param_key, p.param_value)
        FROM curve_fit_params p
        WHERE p.result_id = r.id AND p.param_category = 'fit'
    ) AS fit_params,
    (
        SELECT jsonb_object_agg(p.param_key, COALESCE(p.param_value::text, p.param_text))
        FROM curve_fit_params p
        WHERE p.result_id = r.id AND p.param_category = 'method'
    ) AS method_params
FROM curve_fit_results r
LEFT JOIN curve_fit_metrics m ON r.id = m.result_id;
```

### 4.4 视图使用示例

```sql
-- 示例1：查询泵站1的所有泵组曲线
SELECT * FROM v_group_curves
WHERE station_id = 1;

-- 示例2：查询包含设备ID 101的泵组曲线
SELECT * FROM v_group_curves
WHERE 101 = ANY(pump_combination);

-- 示例3：查询3台泵并联的同构泵组曲线
-- 注意：group_type 存储 GroupProcessingStrategy 枚举值（小写）
SELECT * FROM v_group_curves
WHERE pump_count = 3
  AND group_type = 'homogeneous';

-- 示例4：获取泵站1的最新Q-H泵组曲线
SELECT * FROM v_group_curves_latest
WHERE station_id = 1
  AND curve_type = 'qh';

-- 示例5：对比单泵和泵组曲线精度
SELECT
    '单泵' as curve_source,
    AVG(r_squared) as avg_r2,
    COUNT(*) as curve_count
FROM v_single_pump_curves
WHERE curve_type = 'qh'
UNION ALL
SELECT
    '泵组' as curve_source,
    AVG(r_squared) as avg_r2,
    COUNT(*) as curve_count
FROM v_group_curves
WHERE curve_type = 'qh';
```
