```python
class CurveFittingPipeline:
    """特性曲线拟合管道 - 核心协调器

    注意：不再需要传入 db_session 参数，各组件内部使用
    get_connection() 上下文管理器获取数据库连接。
    """

    def __init__(
        self,
        method_registry: 'MethodRegistry',
        result_storage: 'ResultStorage',
        cache_manager: Optional['CacheManager'] = None,
        batch_processor: Optional['BatchProcessor'] = None,
        parameter_optimizer: Optional['ParameterOptimizer'] = None,
        historical_evaluator: Optional['HistoricalDataEvaluator'] = None,
        time_window_splitter: Optional['TimeWindowSplitter'] = None,
        result_output: Optional['EnhancedResultOutput'] = None
    ):
        """
        初始化拟合管道

        Args:
            method_registry: 方法注册表（必需）
            result_storage: 结果存储器（必需）
            其他参数: 可选组件，不提供则使用默认实现

        注意：不需要传入 db_session，各组件内部使用 get_connection()
        """
        self._logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")

        # 必需组件
        self._method_registry = method_registry
        self._result_storage = result_storage

        # 可选组件（延迟初始化）
        self._cache_manager = cache_manager
        self._batch_processor = batch_processor
        self._parameter_optimizer = parameter_optimizer
        self._historical_evaluator = historical_evaluator
        self._time_window_splitter = time_window_splitter or TimeWindowSplitter()
        self._result_output = result_output

        # 曲线类型到Pipeline的映射
        self._curve_pipelines: Dict[str, 'BaseCurvePipeline'] = {}

        self._logger.info("CurveFittingPipeline 初始化完成")

    def register_curve_pipeline(
        self,
        curve_type: str,
        pipeline: 'BaseCurvePipeline'
    ) -> None:
        """注册曲线类型对应的Pipeline"""
        self._curve_pipelines[curve_type] = pipeline
        self._logger.info(f"注册曲线Pipeline: {curve_type}")
```

#### 2.1.2 `fit()` - 核心拟合流程（10阶段）

```python
    # 🟡-06修复：默认超时时间（秒）
    DEFAULT_FIT_TIMEOUT = 300  # 5分钟
    ML_FIT_TIMEOUT = 600       # ML方法允许更长时间（10分钟）

    def fit(
        self,
        device_id: int,
        curve_type: str,
        start_time: Optional[datetime] = None,
        end_time: Optional[datetime] = None,
        method_id: Optional[str] = None,
        force_refit: bool = False,
        timeout: Optional[int] = None  # 🟡-06修复：添加超时参数
    ) -> 'FitResult':
        """
        执行曲线拟合（10阶段流程，带超时保护）

        Args:
            device_id: 设备ID
            curve_type: 曲线类型 ('qh', 'qp', 'qeta')
            start_time: 拟合数据开始时间（可选，自动确定）
            end_time: 拟合数据结束时间（可选，自动确定）
            method_id: 指定拟合方法（可选，自动选择）
            force_refit: 是否强制重新拟合
            timeout: 超时时间（秒），默认300秒，ML方法默认600秒（🟡-06修复）

        Returns:
            FitResult: 拟合结果对象

        Raises:
            ValueError: 参数无效
            DataNotFoundError: 数据不存在或不足
            FittingError: 拟合失败
            TimeoutError: 拟合超时（🟡-06修复）
        """
        self._logger.info(
            f"开始拟合: device_id={device_id}, curve_type={curve_type}",
            extra={"extra_data": {"设备ID": device_id, "曲线类型": curve_type}}
        )
        
        # 获取对应的曲线Pipeline
        if curve_type not in self._curve_pipelines:
            raise ValueError(f"未注册的曲线类型: {curve_type}")
        curve_pipeline = self._curve_pipelines[curve_type]

        try:
            # ========== 阶段0: 时间窗口划分 ==========
            self._logger.debug("阶段0: 时间窗口划分")
            time_split = self._time_window_splitter.split(
                device_id=device_id,
                curve_type=curve_type
            )

            # 如果未指定时间范围，使用自动划分的拟合窗口
            if start_time is None:
                start_time = time_split.fit_window[0]
            if end_time is None:
                end_time = time_split.fit_window[1]

            # ========== 阶段1: 数据提取 ==========
            self._logger.debug("阶段1: 数据提取")
            raw_data = curve_pipeline._extract_data(
                device_id=device_id,
                start_time=start_time,
                end_time=end_time
            )

            if len(raw_data) < 100:  # 最小数据点要求
                raise DataNotFoundError(
                    f"数据不足: 需要至少100条，实际{len(raw_data)}条"
                )

            # ========== 阶段2: 数据预处理 ==========
            self._logger.debug("阶段2: 数据预处理")
            cleaned_data = curve_pipeline._preprocess(raw_data)

            # ========== 阶段3: 约束计算 ==========
            self._logger.debug("阶段3: 约束计算")
            device_params = self._get_device_params(device_id)
            constraints = curve_pipeline._calculate_constraints(
                data=cleaned_data,
                device_params=device_params
            )

            # ========== 阶段4: 数据归一化 ==========
            self._logger.debug("阶段4: 数据归一化")
            normalizer = curve_pipeline._get_normalizer()
            normalized_data, norm_params = normalizer.fit_transform(
                data=cleaned_data,
                constraints=constraints
            )

            # ========== 阶段5: 方法选择 ==========
            self._logger.debug("阶段5: 方法选择")
            if method_id is None:
                method_id = self._method_registry.select_best_method(
                    curve_type=curve_type,
                    data_characteristics=self._analyze_data(cleaned_data),
                    available_columns=list(cleaned_data.columns)
                )

            method = self._method_registry.get_method(method_id)
            self._logger.info(f"选择方法: {method_id}")

            # ========== 阶段6: 曲线拟合（带超时保护） ==========
            self._logger.debug("阶段6: 曲线拟合")
            X = normalized_data['X'].values
            y = normalized_data['Y'].values

            # 🟡-06修复：确定超时时间
            if timeout is None:
                # ML方法允许更长时间
                is_ml_method = method_id.startswith('ml_') if method_id else False
                timeout = self.ML_FIT_TIMEOUT if is_ml_method else self.DEFAULT_FIT_TIMEOUT

            # 🟡-06修复：带超时的拟合执行
            import concurrent.futures

            def _execute_fit():
                return method.fit(X=X, y=y, constraints=constraints)

            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as executor:
                future = executor.submit(_execute_fit)
                try:
                    method_result = future.result(timeout=timeout)
                except concurrent.futures.TimeoutError:
                    self._logger.error(
                        "[拟合] 超时终止",
                        extra={"extra_data": {
                            "设备ID": device_id,
                            "曲线类型": curve_type,
                            "方法ID": method_id,
                            "超时时间秒": timeout,
                            "影响": "拟合被强制终止",
                            "建议": "尝试减少数据量或使用更简单的方法"
                        }}
                    )
                    raise TimeoutError(
                        f"拟合超时：设备{device_id}的{curve_type}曲线拟合超过{timeout}秒"
                    )

            # ========== 阶段7: 结果验证 ==========
            self._logger.debug("阶段7: 结果验证")
            validation_result = self._validate_fit_result(
                method_result=method_result,
                constraints=constraints,
                curve_type=curve_type
            )

            if not validation_result.is_valid:
                self._logger.warning(
                    f"拟合结果验证失败: {validation_result.reason}"
                )

            # ========== 阶段8: 历史数据评估 ==========
            self._logger.debug("阶段8: 历史数据评估")
            historical_evaluation = None
            if time_split.can_evaluate and self._historical_evaluator:
                # 构建预测函数
                def predict_func(x: np.ndarray) -> np.ndarray:
                    return method.predict(x, method_result.params)

                historical_evaluation = self._historical_evaluator.evaluate_prediction_accuracy(
                    device_id=device_id,
                    curve_type=curve_type,
                    predict_func=predict_func,
                    test_window=time_split.test_window,
                    norm_params=norm_params
                )

            # ========== 阶段9: 构建并存储结果 ==========
            self._logger.debug("阶段9: 存储结果")
            fit_result = FitResult(
                device_id=device_id,
                curve_type=curve_type,
                method_id=method_id,
                params=method_result.params,
                r_squared=method_result.r_squared,
                rmse=method_result.rmse,
                mae=method_result.mae,
                mape=method_result.mape,
                constraints=constraints,
                norm_params=norm_params,
                validation_result=validation_result,
                historical_evaluation=historical_evaluation,
                time_split={
                    'fit_window': [start_time.isoformat(), end_time.isoformat()],
                    'test_window': [
                        time_split.test_window[0].isoformat(),
                        time_split.test_window[1].isoformat()
                    ] if time_split.can_evaluate else None,
                    'fit_ratio': time_split.fit_ratio
                },
                data_points=len(cleaned_data),
                fitted_at=datetime.now()
            )

            # 保存到数据库
            version = self._result_storage.save(
                device_id=device_id,
                curve_type=curve_type,
                fit_result=fit_result
            )
            fit_result.version = version

            self._logger.info(
                f"拟合完成: version={version}, R²={method_result.r_squared:.4f}",
                extra={"extra_data": {
                    "设备ID": device_id,
                    "曲线类型": curve_type,
                    "版本": version,
                    "R²": method_result.r_squared,
                    "RMSE": method_result.rmse
                }}
            )

            return fit_result

        except Exception as e:
            self._logger.error(f"拟合失败: {e}", exc_info=True)
            raise FittingError(f"拟合失败: {e}") from e

    def _get_device_params(self, device_id: int) -> Dict[str, Any]:
        """
        获取设备额定参数

        从 device_rated_params 表读取设备额定参数（键值对存储）。
        **强制规则**：
        1. 必须从 device_rated_params 表读取，不允许使用硬编码或默认值
        2. 参数不存在时必须输出错误日志并抛出异常
        3. 禁止使用任何默认值

        Args:
            device_id: 设备ID

        Returns:
            Dict[str, Any]: 设备参数字典，包含 rated_flow, rated_head, rated_power, rated_efficiency

        Raises:
            ValueError: 设备参数缺失时抛出
        """
        # 必需的参数列表
        required_params = ['rated_flow', 'rated_head', 'rated_power']
        optional_params = ['rated_efficiency']  # 可选，可计算得出

        # 从 device_rated_params 表读取（使用 get_connection()）
        sql = """
            SELECT param_key, value_numeric, unit
            FROM device_rated_params
            WHERE device_id = %s
              AND param_key IN ('rated_flow', 'rated_head', 'rated_power', 'rated_efficiency')
              AND (effective_to IS NULL OR effective_to > NOW())
            ORDER BY param_key
        """

        with get_connection() as conn:
            with conn.cursor() as cursor:
                cursor.execute(sql, (device_id,))
                rows = cursor.fetchall()

        # 转换为字典
        params = {}
        for row in rows:
            params[row[0]] = row[1]  # param_key, value_numeric

        # 检查必需参数是否存在
        missing_params = [p for p in required_params if p not in params or params[p] is None]
        if missing_params:
            self._logger.error(
                "设备额定参数缺失，无法进行特性曲线拟合",
                extra={"extra_data": {
                    "device_id": device_id,
                    "missing_params": missing_params,
                    "action": "abort_fitting",
                    "suggestion": "请在device_rated_params表中添加缺失的额定参数"
                }}
            )
            raise ValueError(f"设备{device_id}缺少必需的额定参数: {missing_params}")

        # 如果 rated_efficiency 不存在，计算得出
        # 公式：η = (ρ * g * Q * H) / (P * 1000)
        if 'rated_efficiency' not in params or params['rated_efficiency'] is None:
            rho = 1000  # 水密度 kg/m³
            g = 9.81    # 重力加速度 m/s²
            Q = params['rated_flow'] / 3600  # m³/h → m³/s
            H = params['rated_head']  # m
            P = params['rated_power']  # kW

            if P > 0:
                params['rated_efficiency'] = (rho * g * Q * H) / (P * 1000)
                self._logger.info(
                    f"计算得出额定效率: {params['rated_efficiency']:.4f}",
                    extra={"extra_data": {"device_id": device_id}}
                )
            else:
                self._logger.warning(
                    f"无法计算额定效率：额定功率为0",
                    extra={"extra_data": {"device_id": device_id}}
                )

        return params

    def _analyze_data(self, data: pd.DataFrame) -> Dict[str, Any]:
        """分析数据特征"""
        return {
            'point_count': len(data),
            'has_outliers': False,  # 预处理后应无异常值
            'coverage': 'full',
            'noise_level': 'low'
        }

    def _validate_fit_result(
        self,
        method_result: 'MethodResult',
        constraints: Dict[str, Any],
        curve_type: str
    ) -> 'ValidationResult':
        """验证拟合结果"""
        is_valid = True
        reasons = []

        # 检查 R²
        if method_result.r_squared < 0.9:
            is_valid = False
            reasons.append(f"R²={method_result.r_squared:.4f} < 0.9")

        # 检查 RMSE（相对于数据范围）
        if method_result.rmse > constraints.get('max_rmse', float('inf')):
            is_valid = False
            reasons.append(f"RMSE过大: {method_result.rmse:.4f}")

        return ValidationResult(
            is_valid=is_valid,
            reason='; '.join(reasons) if reasons else None,
            r_squared=method_result.r_squared,
            rmse=method_result.rmse
        )
```

```python
        self,
```

```python
    def compare_versions(
        self,
        device_id: int,
        curve_type: str,
        version1: str,
        version2: str
    ) -> Dict[str, Any]:
        """
        对比两个版本的拟合结果

        Args:
            device_id: 设备ID
            curve_type: 曲线类型
            version1: 版本1
            version2: 版本2

        Returns:
            Dict: 对比结果
        """
        result1 = self._result_storage.load(device_id, curve_type, version1)
        result2 = self._result_storage.load(device_id, curve_type, version2)

        if result1 is None or result2 is None:
            raise ValueError("版本不存在")

        # 指标对比
        metrics_diff = {
            'r_squared': {
                'v1': result1.r_squared,
                'v2': result2.r_squared,
                'diff': result2.r_squared - result1.r_squared,
                'improved': result2.r_squared > result1.r_squared
            },
            'rmse': {
                'v1': result1.rmse,
                'v2': result2.rmse,
                'diff': result2.rmse - result1.rmse,
                'improved': result2.rmse < result1.rmse
            },
            'mape': {
                'v1': result1.mape,
                'v2': result2.mape,
                'diff': result2.mape - result1.mape,
                'improved': result2.mape < result1.mape
            }
        }

        # 参数对比
        param_changes = {}
        for key in result1.params:
            if key in result2.params:
                v1, v2 = result1.params[key], result2.params[key]
                param_changes[key] = {
                    'v1': v1,
                    'v2': v2,
                    'change_percent': ((v2 - v1) / v1 * 100) if v1 != 0 else None
                }

        # 方法变化
        method_changed = result1.method_id != result2.method_id

        return {
            'version1': version1,
            'version2': version2,
            'metrics_diff': metrics_diff,
            'param_changes': param_changes,
            'method_changed': method_changed,
            'overall_improved': (
                metrics_diff['r_squared']['improved'] and
                metrics_diff['rmse']['improved']
            )
        }
```

### 2.2 MethodRegistry（方法注册表）

```python
class MethodRegistry:
    """拟合方法注册表 - 管理所有可用的拟合方法"""

    def __init__(self):
        self._logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")
        # 结构: {curve_type: {method_id: MethodInfo}}
        self._methods: Dict[str, Dict[str, 'MethodInfo']] = {
            'qh': {},
            'qp': {},
            'qeta': {}
        }

    def register(
        self,
        curve_type: str,
        method_id: str,
        method_name: str,
        method_class: type,
        priority: int = 100,
        description: str = "",
        dependencies: Optional[List[str]] = None,
        applicable_conditions: Optional[Dict[str, Any]] = None
    ) -> None:
        """
        注册拟合方法

        Args:
            curve_type: 曲线类型 ('qh', 'qp', 'qeta')
            method_id: 方法唯一标识（如 'qh_poly2'）
            method_name: 方法显示名称（中文）
            method_class: 方法类（BaseMethod的子类）
            priority: 优先级（越高越优先，默认100）
            description: 方法描述
            dependencies: 依赖的数据列列表
            applicable_conditions: 适用条件字典
        """
        if curve_type not in self._methods:
            raise ValueError(f"无效的曲线类型: {curve_type}")

        method_info = MethodInfo(
            method_id=method_id,
            method_name=method_name,
            method_class=method_class,
            priority=priority,
            description=description,
            dependencies=dependencies or [],
            applicable_conditions=applicable_conditions or {}
        )

        self._methods[curve_type][method_id] = method_info
        self._logger.info(f"注册方法: {curve_type}/{method_id} (优先级: {priority})")

    def get_method(self, method_id: str) -> 'BaseMethod':
        """
        获取方法实例

        Args:
            method_id: 方法ID（格式: 'qh_poly2' 或完整ID）

        Returns:
            BaseMethod: 方法实例

        Raises:
            ValueError: 方法不存在
        """
        # 解析 curve_type 和 method_id
        for curve_type, methods in self._methods.items():
            if method_id in methods:
                method_info = methods[method_id]
                return method_info.method_class()

        raise ValueError(f"方法不存在: {method_id}")

    def select_best_method(
        self,
        curve_type: str,
        data_characteristics: Dict[str, Any],
        available_columns: List[str]
    ) -> str:
        """
        选择最佳拟合方法

        Args:
            curve_type: 曲线类型
            data_characteristics: 数据特征
            available_columns: 可用数据列

        Returns:
            str: 选中的方法ID

        Raises:
            ValueError: 无适用方法
        """
        if curve_type not in self._methods:
            raise ValueError(f"无效的曲线类型: {curve_type}")

        candidates = []

        for method_id, method_info in self._methods[curve_type].items():
            # 检查依赖列是否可用
            if not all(col in available_columns for col in method_info.dependencies):
                self._logger.debug(f"方法 {method_id} 缺少依赖列")
                continue

            # 检查适用条件
            conditions_met = True
            for condition_key, condition_value in method_info.applicable_conditions.items():
                if condition_key in data_characteristics:
                    if data_characteristics[condition_key] != condition_value:
                        conditions_met = False
                        break

            if conditions_met:
                candidates.append((method_info.priority, method_id))

        if not candidates:
            raise ValueError(f"无适用方法: curve_type={curve_type}")

        # 按优先级排序，返回最高优先级
        candidates.sort(reverse=True)
        selected = candidates[0][1]

        self._logger.info(f"选择方法: {selected} (从{len(candidates)}个候选中)")
        return selected
```

```python
@dataclass
class TimeWindowSplitResult:
    """时间窗口划分结果"""
    fit_window: Tuple[datetime, datetime]    # 拟合窗口
    test_window: Tuple[datetime, datetime]   # 测试窗口
    fit_ratio: float                         # 拟合数据占比
    can_evaluate: bool                       # 是否有足够测试数据
    total_days: int                          # 总天数
    fit_days: int                            # 拟合天数
    test_days: int                           # 测试天数


class TimeWindowSplitter:
    """时间窗口划分器 - 自动划分拟合/测试数据时间范围"""

    # 默认配置
    DEFAULT_FIT_RATIO = 0.75          # 拟合数据占75%
    DEFAULT_MIN_FIT_DAYS = 7          # 最少拟合天数
    DEFAULT_MIN_TEST_DAYS = 2         # 最少测试天数
    DEFAULT_MIN_TEST_POINTS = 100     # 最少测试数据点

    # 曲线类型对应的指标
    CURVE_METRICS = {
        'qh': ['pump_flow_rate', 'pump_head'],
        'qp': ['pump_flow_rate', 'pump_active_power'],
        'qeta': ['pump_flow_rate', 'pump_efficiency']
    }

    def __init__(
        self,
        fit_ratio: float = DEFAULT_FIT_RATIO,
        min_fit_days: int = DEFAULT_MIN_FIT_DAYS,
        min_test_days: int = DEFAULT_MIN_TEST_DAYS,
        min_test_points: int = DEFAULT_MIN_TEST_POINTS
    ):
        """初始化时间窗口划分器（不需要传入db_session）"""
        self._logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")
        self._fit_ratio = fit_ratio
        self._min_fit_days = min_fit_days
        self._min_test_days = min_test_days
        self._min_test_points = min_test_points

    def split(
        self,
        device_id: int,
        curve_type: str
    ) -> TimeWindowSplitResult:
        """
        自动划分时间窗口

        Args:
            device_id: 设备ID
            curve_type: 曲线类型 ('qh', 'qp', 'qeta')

        Returns:
            TimeWindowSplitResult: 划分结果

        处理流程:
            1. 查询 fact_measurements 表获取数据时间范围
            2. 计算总天数，按 75%/25% 比例划分
            3. 验证测试数据量是否足够
            4. 返回划分结果
        """
        # 查询数据时间范围
        data_range = self._query_data_range(device_id, curve_type)

        if data_range is None:
            raise ValueError(f"设备 {device_id} 无 {curve_type} 曲线数据")

        min_time, max_time, total_points = data_range

        # 计算总天数
        total_days = (max_time - min_time).days

        if total_days < self._min_fit_days:
            raise ValueError(
                f"数据时间跨度不足: {total_days}天 < {self._min_fit_days}天"
            )

        # 按比例划分
        fit_days = int(total_days * self._fit_ratio)
        test_days = total_days - fit_days

        # 确保最小天数
        if fit_days < self._min_fit_days:
            fit_days = self._min_fit_days
            test_days = total_days - fit_days

        # 计算分割点
        split_time = min_time + timedelta(days=fit_days)

        # 检查测试数据是否足够
        can_evaluate = (
            test_days >= self._min_test_days and
            total_points * (1 - self._fit_ratio) >= self._min_test_points
        )

        result = TimeWindowSplitResult(
            fit_window=(min_time, split_time),
            test_window=(split_time, max_time),
            fit_ratio=self._fit_ratio,
            can_evaluate=can_evaluate,
            total_days=total_days,
            fit_days=fit_days,
            test_days=test_days
        )

        self._logger.info(
            f"时间窗口划分: fit={fit_days}天, test={test_days}天, can_evaluate={can_evaluate}"
        )

        return result

    def _query_data_range(
        self,
        device_id: int,
        curve_type: str
    ) -> Optional[Tuple[datetime, datetime, int]]:
        """
        查询数据时间范围

        Returns:
            Tuple[min_time, max_time, total_points] or None
        """
        metrics = self.CURVE_METRICS.get(curve_type)
        if not metrics:
            raise ValueError(f"无效的曲线类型: {curve_type}")

        # 构建 metric_key 列表（使用 psycopg3 的参数占位符）
        # 注意：IN 子句需要特殊处理
        placeholders = ", ".join([f"%s" for _ in metrics])

        sql = f"""
            SELECT
                MIN(fm.ts_bucket) as min_time,
                MAX(fm.ts_bucket) as max_time,
                COUNT(DISTINCT fm.ts_bucket) as total_points
            FROM fact_measurements fm
            JOIN dim_metric_config mc ON mc.id = fm.metric_id
            WHERE fm.device_id = %s
              AND mc.metric_key IN ({placeholders})
        """

        with get_connection() as conn:
            with conn.cursor() as cursor:
                cursor.execute(sql, (device_id, *metrics))
                row = cursor.fetchone()

        if row is None or row[0] is None:
            return None

        return (row[0], row[1], row[2])
```

### 2.5 HistoricalDataEvaluator（历史数据评估器）

```python
class HistoricalDataEvaluator:
    """历史数据评估器 - 评估曲线对独立测试数据的预测能力

    使用 get_connection() 上下文管理器获取数据库连接。
    """

    # 通过标准
    PASS_THRESHOLD_5_PERCENT = 0.90   # 90%数据点偏差<5%
    PASS_THRESHOLD_10_PERCENT = 0.98  # 98%数据点偏差<10%

    # 曲线类型对应的指标
    CURVE_METRICS = {
        'qh': {'x': 'pump_flow_rate', 'y': 'pump_head'},
        'qp': {'x': 'pump_flow_rate', 'y': 'pump_active_power'},
        'qeta': {'x': 'pump_flow_rate', 'y': 'pump_efficiency'}
    }

    def __init__(self):
        """初始化历史数据评估器（不需要传入db_session）"""
        self._logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")

    def evaluate_prediction_accuracy(
        self,
        device_id: int,
        curve_type: str,
        predict_func: Callable[[np.ndarray], np.ndarray],
        test_window: Tuple[datetime, datetime],
        norm_params: Dict[str, float]
    ) -> Dict[str, Any]:
        """
        评估曲线预测能力

        Args:
            device_id: 设备ID
            curve_type: 曲线类型
            predict_func: 预测函数（归一化空间）
            test_window: 测试时间窗口 (start, end)
            norm_params: 归一化参数（用于转换）

        Returns:
            Dict: 评估结果
        """
        self._logger.info(
            f"开始预测评估: device={device_id}, curve={curve_type}, "
            f"window={test_window[0]} ~ {test_window[1]}"
        )

        # 1. 加载测试数据
        test_data = self._load_test_data(
            device_id=device_id,
            curve_type=curve_type,
            start_time=test_window[0],
            end_time=test_window[1]
        )

        if len(test_data) == 0:
            self._logger.warning("测试数据为空")
            return {
                'overall_passed': False,
                'reason': '测试数据为空',
                'test_points': 0
            }

        # 2. 归一化测试数据
        X_raw = test_data['X'].values
        Y_actual = test_data['Y'].values

        X_norm = X_raw / norm_params.get('X_scale', 1.0)
        Y_scale = norm_params.get('Y_scale', 1.0)

        # 3. 预测
        Y_pred_norm = predict_func(X_norm)
        Y_pred = Y_pred_norm * Y_scale

        # 4. 计算偏差
        # 相对偏差 = |预测值 - 实际值| / 实际值 * 100%
        with np.errstate(divide='ignore', invalid='ignore'):
            relative_deviation = np.abs(Y_pred - Y_actual) / np.abs(Y_actual) * 100
            # 处理除零情况
            relative_deviation = np.where(
                np.isfinite(relative_deviation),
                relative_deviation,
                0.0
            )

        # 5. 统计偏差分布
        deviation_stats = {
            'mean': float(np.mean(relative_deviation)),
            'std': float(np.std(relative_deviation)),
            'max': float(np.max(relative_deviation)),
            'min': float(np.min(relative_deviation)),
            'p50': float(np.percentile(relative_deviation, 50)),
            'p90': float(np.percentile(relative_deviation, 90)),
            'p95': float(np.percentile(relative_deviation, 95)),
            'p99': float(np.percentile(relative_deviation, 99))
        }

        # 6. 计算通过率
        within_5_percent = np.sum(relative_deviation <= 5) / len(relative_deviation)
        within_10_percent = np.sum(relative_deviation <= 10) / len(relative_deviation)

        pass_rate = {
            'within_5_percent': float(within_5_percent),
            'within_10_percent': float(within_10_percent)
        }

        # 7. 分段评估（低/中/高流量）
        segment_evaluation = self._evaluate_segments(
            X_raw, relative_deviation, norm_params
        )

        # 8. 判断是否通过
        overall_passed = (
            within_5_percent >= self.PASS_THRESHOLD_5_PERCENT and
            within_10_percent >= self.PASS_THRESHOLD_10_PERCENT
        )

        result = {
            'overall_passed': overall_passed,
            'test_points': len(test_data),
            'deviation_stats': deviation_stats,
            'pass_rate': pass_rate,
            'segment_evaluation': segment_evaluation,
            'test_window': {
                'start': test_window[0].isoformat(),
                'end': test_window[1].isoformat()
            }
        }

        self._logger.info(
            f"预测评估完成: passed={overall_passed}, "
            f"within_5%={within_5_percent:.2%}, within_10%={within_10_percent:.2%}"
        )

        return result

    def _load_test_data(
        self,
        device_id: int,
        curve_type: str,
        start_time: datetime,
        end_time: datetime
    ) -> pd.DataFrame:
        """加载测试数据（使用 get_connection() 获取连接）"""
        metrics = self.CURVE_METRICS.get(curve_type)
        if not metrics:
            raise ValueError(f"无效的曲线类型: {curve_type}")

        # 使用 get_connection() 上下文管理器
        sql = f"""
            SELECT
                ts_bucket,
                MAX(CASE WHEN metric_id = (
                    SELECT id FROM dim_metric_config WHERE metric_key = %s
                ) THEN value END) AS X,
                MAX(CASE WHEN metric_id = (
                    SELECT id FROM dim_metric_config WHERE metric_key = %s
                ) THEN value END) AS Y
            FROM fact_measurements
            WHERE device_id = %s
              AND ts_bucket BETWEEN %s AND %s
            GROUP BY ts_bucket
            HAVING
                MAX(CASE WHEN metric_id = (
                    SELECT id FROM dim_metric_config WHERE metric_key = %s
                ) THEN value END) IS NOT NULL
                AND MAX(CASE WHEN metric_id = (
                    SELECT id FROM dim_metric_config WHERE metric_key = %s
                ) THEN value END) IS NOT NULL
            ORDER BY ts_bucket
        """

        with get_connection() as conn:
            with conn.cursor() as cursor:
                cursor.execute(sql, (
                    metrics["x"], metrics["y"],
                    device_id, start_time, end_time,
                    metrics["x"], metrics["y"]
                ))
                rows = cursor.fetchall()
                columns = [desc[0] for desc in cursor.description]

        return pd.DataFrame(rows, columns=columns)

    def _evaluate_segments(
        self,
        X: np.ndarray,
        deviations: np.ndarray,
        norm_params: Dict[str, float]
    ) -> Dict[str, Dict[str, float]]:
        """分段评估（低/中/高流量）"""
        X_rated = norm_params.get('X_scale', np.max(X))

        # 分段边界
        low_mask = X <= 0.33 * X_rated
        mid_mask = (X > 0.33 * X_rated) & (X <= 0.66 * X_rated)
        high_mask = X > 0.66 * X_rated

        segments = {}

        for name, mask in [('low', low_mask), ('mid', mid_mask), ('high', high_mask)]:
            if np.sum(mask) > 0:
                seg_dev = deviations[mask]
                segments[name] = {
                    'count': int(np.sum(mask)),
                    'mean_deviation': float(np.mean(seg_dev)),
                    'max_deviation': float(np.max(seg_dev)),
                    'within_5_percent': float(np.sum(seg_dev <= 5) / len(seg_dev))
                }
            else:
                segments[name] = {'count': 0, 'mean_deviation': None}

        return segments

    def evaluate_data_quality(
        self,
        data: pd.DataFrame,
        required_columns: List[str]
    ) -> Dict[str, Any]:
        """
        评估数据质量

        Args:
            data: 输入数据
            required_columns: 必需列列表

        Returns:
            Dict: 质量评估结果
        """
        total_rows = len(data)

        # 完整性检查
        missing_counts = {}
        for col in required_columns:
            if col in data.columns:
                missing_counts[col] = int(data[col].isna().sum())
            else:
                missing_counts[col] = total_rows

        completeness = 1 - sum(missing_counts.values()) / (total_rows * len(required_columns))

        # 异常值检查（3σ原则）
        outlier_counts = {}
        for col in required_columns:
            if col in data.columns:
                values = data[col].dropna()
                mean, std = values.mean(), values.std()
                outliers = ((values < mean - 3*std) | (values > mean + 3*std)).sum()
                outlier_counts[col] = int(outliers)

        outlier_ratio = sum(outlier_counts.values()) / (total_rows * len(required_columns))

        return {
            'total_rows': total_rows,
            'completeness': float(completeness),
            'missing_counts': missing_counts,
            'outlier_ratio': float(outlier_ratio),
            'outlier_counts': outlier_counts,
            'quality_score': float(completeness * (1 - outlier_ratio))
        }
```

```python
class DeviceParamsReader:
    """设备参数读取器 - 从数据库读取和管理设备参数

    使用 get_connection() 上下文管理器获取数据库连接。

    # 缓存过期时间（秒）
    CACHE_TTL = 300

    def __init__(self, use_cache: bool = True):
        """初始化设备参数读取器（不需要传入db_session）"""
        self._logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")
        self._use_cache = use_cache
        self._cache: Dict[int, Tuple[Dict[str, Any], datetime]] = {}

    def get_device_params(
        self,
        device_id: int,
        force_refresh: bool = False
    ) -> Dict[str, Any]:
        """
        获取设备参数

        Args:
            device_id: 设备ID
            force_refresh: 是否强制刷新缓存

        Returns:
            Dict: 设备参数

        Raises:
            ValueError: 设备不存在
        """
        # 检查缓存
        if self._use_cache and not force_refresh:
            if device_id in self._cache:
                params, cached_at = self._cache[device_id]
                if (datetime.now() - cached_at).seconds < self.CACHE_TTL:
                    return params

        # 使用 get_connection() 上下文管理器
        sql = """
            SELECT
                id, name, device_type, model,
                rated_flow, rated_head, rated_power, rated_efficiency,
                rated_speed, max_flow, min_flow, shutoff_head,
                specific_speed, impeller_diameter,
                installation_date, manufacturer,
                extra_params
            FROM dim_device
            WHERE id = %s
        """

        with get_connection() as conn:
            with conn.cursor() as cursor:
                cursor.execute(sql, (device_id,))
                row = cursor.fetchone()
                if row:
                    columns = [desc[0] for desc in cursor.description]

        if row is None:
            raise ValueError(f"设备不存在: {device_id}")

        params = dict(zip(columns, row))

        # 解析 extra_params JSONB
        if params.get('extra_params'):
            extra = json.loads(params['extra_params']) if isinstance(params['extra_params'], str) else params['extra_params']
            params.update(extra)

        # 更新缓存
        if self._use_cache:
            self._cache[device_id] = (params, datetime.now())

        return params

    def set_device_params(
        self,
        device_id: int,
        params: Dict[str, Any],
        merge: bool = True
    ) -> None:
        """
        设置设备参数（使用 get_connection() 获取连接）

        Args:
            device_id: 设备ID
            params: 要设置的参数
            merge: 是否合并现有参数（True=合并，False=覆盖）
        """
        # 分离标准字段和扩展字段
        standard_fields = {
            'rated_flow', 'rated_head', 'rated_power', 'rated_efficiency',
            'rated_speed', 'max_flow', 'min_flow', 'shutoff_head',
            'specific_speed', 'impeller_diameter'
        }

        standard_updates = {k: v for k, v in params.items() if k in standard_fields}
        extra_updates = {k: v for k, v in params.items() if k not in standard_fields}

        with get_connection() as conn:
            with conn.cursor() as cursor:
                # 更新标准字段
                if standard_updates:
                    set_clauses = ', '.join([f"{k} = %s" for k in standard_updates.keys()])
                    sql = f"""
                        UPDATE dim_device
                        SET {set_clauses}, updated_at = %s
                        WHERE id = %s
                    """
                    cursor.execute(sql, (*standard_updates.values(), datetime.now(), device_id))

                # 更新扩展字段
                if extra_updates:
                    if merge:
                        # 合并模式：使用 jsonb ||
                        sql = """
                            UPDATE dim_device
                            SET extra_params = COALESCE(extra_params, '{}'::jsonb) || %s::jsonb,
                                updated_at = %s
                            WHERE id = %s
                        """
                    else:
                        # 覆盖模式
                        sql = """
                            UPDATE dim_device
                            SET extra_params = %s::jsonb,
                                updated_at = %s
                            WHERE id = %s
                        """
                    cursor.execute(sql, (
                        json.dumps(extra_updates, ensure_ascii=False),
                        datetime.now(),
                        device_id
                    ))

            conn.commit()

        # 清除缓存
        self.invalidate_cache(device_id)

        self._logger.info(f"更新设备参数: device_id={device_id}")

    def get_curve_constraints(
        self,
        device_id: int,
        curve_type: str
    ) -> Dict[str, Any]:
        """
        获取曲线约束

        Args:

        Returns:

            'rated_flow': params.get('rated_flow'),

        }

        if curve_type == 'qh':
            constraints.update({

                'shutoff_head': params.get('shutoff_head'),
                # Q-H曲线约束：H在Q增加时单调递减
                'monotonic_decreasing': True
            })
        elif curve_type == 'qp':
            constraints.update({
                'rated_power': params.get('rated_power'),
                # Q-P曲线可能非单调
                'monotonic_decreasing': False
            })
        elif curve_type == 'qeta':
            constraints.update({
                'rated_efficiency': params.get('rated_efficiency'),
                # Q-η曲线在额定点达到最大
                'has_maximum': True,
                'max_at_rated': True
            })

        return constraints
```

### 2.7 DeviceParamsLearner（设备参数学习器）

```python
class DeviceParamsLearner:
    """设备参数学习器 - 从历史数据学习设备参数

    使用 get_connection() 上下文管理器获取数据库连接。
    """

    def __init__(
        self,
        device_params_reader: DeviceParamsReader
    ):
        """初始化设备参数学习器（不需要传入db_session）"""
        self._logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")
        self._params_reader = device_params_reader

    def learn_from_history(
        self,
        device_id: int,
        start_time: Optional[datetime] = None,
        end_time: Optional[datetime] = None,
        min_points: int = 1000
    ) -> Dict[str, Any]:
        """
        从历史数据学习设备参数

        Args:
            device_id: 设备ID
            start_time: 开始时间（可选，默认全部历史）
            end_time: 结束时间（可选）
            min_points: 最小数据点要求

        Returns:
            Dict: 学习到的参数
        """
        self._logger.info(f"开始从历史数据学习参数: device_id={device_id}")

        # 构建时间条件和参数
        time_conditions = []
        query_params = [device_id]

        if start_time:
            time_conditions.append("AND ts_bucket >= %s")
            query_params.append(start_time)
        if end_time:
            time_conditions.append("AND ts_bucket <= %s")
            query_params.append(end_time)

        time_condition = " ".join(time_conditions)

        # 查询历史数据统计（使用 get_connection()）
        sql = f"""
            SELECT
                mc.metric_key,
                COUNT(*) as count,
                AVG(fm.value) as avg_value,
                STDDEV(fm.value) as std_value,
                MIN(fm.value) as min_value,
                MAX(fm.value) as max_value,
                PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY fm.value) as p50,
                PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY fm.value) as p95,
                PERCENTILE_CONT(0.99) WITHIN GROUP (ORDER BY fm.value) as p99
            FROM fact_measurements fm
            JOIN dim_metric_config mc ON mc.id = fm.metric_id
            WHERE fm.device_id = %s
              AND mc.metric_key IN (
                  'pump_flow_rate', 'pump_head', 'pump_active_power',
                  'pump_efficiency', 'pump_speed'
              )
              {time_condition}
            GROUP BY mc.metric_key
        """

        with get_connection() as conn:
            with conn.cursor() as cursor:
                cursor.execute(sql, query_params)
                rows = cursor.fetchall()
                columns = [desc[0] for desc in cursor.description]

        if not rows:
            raise ValueError(f"无历史数据: device_id={device_id}")

        # 转换为字典
        stats = {}
        for row in rows:
            row_dict = dict(zip(columns, row))
            stats[row_dict['metric_key']] = row_dict

        # 检查数据量
        min_count = min(s['count'] for s in stats.values())
        if min_count < min_points:
            self._logger.warning(f"数据量不足: {min_count} < {min_points}")

        # 估算参数
        learned_params = {}

        # 1. 额定点估算
        rated_point = self.estimate_rated_point(stats)
        learned_params.update(rated_point)

        # 2. 运行范围估算
        operating_range = self.estimate_operating_range(stats)
        learned_params.update(operating_range)

        # 3. 计算置信度
        confidence = self.calculate_confidence(stats, min_points)
        learned_params['confidence'] = confidence

        self._logger.info(f"参数学习完成: confidence={confidence:.2%}")

        return learned_params

    def estimate_rated_point(
        self,
        stats: Dict[str, Dict[str, Any]]
    ) -> Dict[str, float]:
        """
        估算额定点

        思路：
        - 额定效率 = 效率的最大值或高百分位数
        - 额定流量 = 效率最大时对应的流量
        - 额定扬程 = 额定流量对应的扬程
        - 额定功率 = 额定流量对应的功率
        """
        rated = {}

        # 效率
        if 'pump_efficiency' in stats:
            eta_stats = stats['pump_efficiency']
            # 使用P95作为额定效率估计（避免异常值影响）
            rated['rated_efficiency'] = float(eta_stats['p95'])

        # 流量
        if 'pump_flow_rate' in stats:
            q_stats = stats['pump_flow_rate']
            # 额定流量通常在中高区间
            rated['rated_flow'] = float(q_stats['p50'])

        # 扬程
        if 'pump_head' in stats:
            h_stats = stats['pump_head']
            rated['rated_head'] = float(h_stats['avg_value'])

        # 功率
        if 'pump_active_power' in stats:
            p_stats = stats['pump_active_power']
            rated['rated_power'] = float(p_stats['p50'])

        return rated

    def estimate_operating_range(
        self,
        stats: Dict[str, Dict[str, Any]]
    ) -> Dict[str, float]:
        """
        估算运行范围
        """
        range_params = {}

        if 'pump_flow_rate' in stats:
            q_stats = stats['pump_flow_rate']
            # 最小/最大流量使用统计边界
            range_params['min_flow'] = float(q_stats['min_value'])
            range_params['max_flow'] = float(q_stats['max_value'])
            # 常用范围
            range_params['typical_flow_min'] = float(q_stats['p50'] - 2 * q_stats['std_value'])
            range_params['typical_flow_max'] = float(q_stats['p95'])

        if 'pump_head' in stats:
            h_stats = stats['pump_head']
            range_params['min_head'] = float(h_stats['min_value'])
            range_params['max_head'] = float(h_stats['max_value'])
            # 零流量扬程（shutoff head）估计
            range_params['shutoff_head'] = float(h_stats['p99'])

        return range_params

    def calculate_confidence(
        self,
        stats: Dict[str, Dict[str, Any]],
        min_points: int
    ) -> float:
        """
        计算置信度

        因素：
        - 数据量
        - 数据分布覆盖度
        - 数据稳定性
        """
        scores = []

        for metric_key, metric_stats in stats.items():
            count = metric_stats['count']
            std = metric_stats['std_value'] or 0
            avg = metric_stats['avg_value'] or 1

            # 数据量评分（0-1）
            count_score = min(1.0, count / (min_points * 10))

            # 稳定性评分（CV越小越好）
            cv = abs(std / avg) if avg != 0 else 1.0
            stability_score = max(0, 1 - cv)

            # 覆盖度评分（范围/均值的比例）
            range_ratio = (metric_stats['max_value'] - metric_stats['min_value']) / avg if avg != 0 else 0
            coverage_score = min(1.0, range_ratio)

            # 综合评分
            metric_score = (count_score * 0.4 + stability_score * 0.3 + coverage_score * 0.3)
            scores.append(metric_score)

        return sum(scores) / len(scores) if scores else 0.0
```

```python
from dataclasses import dataclass, field
from typing import Dict, List, Any, Optional, Tuple
from datetime import datetime


@dataclass
class FitResult:
    """拟合结果

    🔴 术语统一：使用 coefficients 而非 params/fit_params
    """
    device_id: int
    curve_type: str
    method_id: str
    coefficients: Dict[str, float]  # 🔴 术语统一：拟合系数
    r_squared: float
    rmse: float
    mae: float
    mape: float
    physics_constraints: Dict[str, Any]  # 🔴 术语统一：物理约束
    norm_params: Dict[str, float]  # 归一化参数（Q_min, Q_max等）
    validation_result: Optional['ValidationResult'] = None
    historical_evaluation: Optional[Dict[str, Any]] = None
    time_split: Optional[Dict[str, Any]] = None
    data_points: int = 0
    fitted_at: datetime = field(default_factory=datetime.now)
    version: Optional[str] = None


@dataclass
class ValidationResult:
    """验证结果"""
    is_valid: bool
    reason: Optional[str] = None
    r_squared: Optional[float] = None
    rmse: Optional[float] = None
    deviation_stats: Optional[Dict[str, float]] = None
    pass_rate: Optional[Dict[str, float]] = None


@dataclass
class MethodResult:
    """方法执行结果

    🔴 术语统一：使用 coefficients 而非 params/fit_params
    """
    coefficients: Dict[str, float]  # 🔴 术语统一：拟合系数
    r_squared: float
    rmse: float
    mae: float
    mape: float
    residuals: Optional[np.ndarray] = None
    converged: bool = True

@dataclass
class BatchResult:
```
