```python
            params = self._load_from_database(device_id)

                "[参数管理] 更新设备参数",
                extra={"extra_data": {

                    "更新值": params
                }}

            # 保存到数据库
            # ... 数据库更新逻辑 ...

            # 更新缓存
            if device_id in self._cache:
                self._cache[device_id].update(params)

            duration = (time.time() - start_time) * 1000

            self._logger.info(
                "[参数管理] 参数更新完成",
                extra={"extra_data": {
                    "设备ID": device_id,
                    "参数来源": source,
                    "耗时ms": duration
                }}
```

### 3.3 ConstraintCache（约束缓存）

#### 3.3.1 功能概述

| 属性 | 说明 |
|------|------|
| **文件路径** | `app/services/characteristic_curves/constraints/constraint_cache.py` |
| **职责** | 缓存约束计算结果，避免重复计算 |
| **设计模式** | 缓存模式（Cache Pattern） |
| **依赖** | hashlib |

**核心职责**：
1. 缓存约束计算结果
2. 管理缓存失效
3. 提供缓存统计

**缓存策略**：
- **缓存键**：`(device_id, curve_type, rated_params_hash)`
- **失效条件**：额定值更新时
- **存储方式**：内存缓存（进程级别）
- **缓存范围**：同一设备、同一曲线类型、相同额定参数

**设计理念**：
- 约束计算成本不高，但频繁调用时可以优化
- 对于同一设备、同一时间窗口的数据，约束参数基本不变
- 缓存失效条件明确：额定值更新时

#### 3.3.2 核心接口

```python
from typing import Dict, Any, Optional
import hashlib
import json
import logging

class ConstraintCache:
    """约束缓存 - 缓存约束计算结果"""

    _cache: Dict[tuple, Dict[str, Any]] = {}
    _stats: Dict[str, int] = {'hits': 0, 'misses': 0}
    _logger = logging.getLogger(__name__)

    @classmethod
    def get(
        cls,
        device_id: int,
        curve_type: str,
        rated_params: Dict[str, Any]
    ) -> Optional[Dict[str, Any]]:
        """
        获取缓存的约束参数

        Args:
            device_id: 设备ID
            curve_type: 曲线类型
            rated_params: 额定参数

        Returns:
            Optional[Dict]: 缓存的约束参数，如果不存在返回None
        """
        key = cls._make_key(device_id, curve_type, rated_params)

        if key in cls._cache:
            cls._stats['hits'] += 1
            cls._logger.debug(
                f"约束缓存命中: device_id={device_id}, curve_type={curve_type}"
            )
            return cls._cache[key]
        else:
            cls._stats['misses'] += 1
            cls._logger.debug(
                f"约束缓存未命中: device_id={device_id}, curve_type={curve_type}"
            )
            return None

    @classmethod
    def set(
        cls,
        device_id: int,
        curve_type: str,
        rated_params: Dict[str, Any],
        constraints: Dict[str, Any]
    ) -> None:
        """
        设置缓存

        Args:
            device_id: 设备ID
            curve_type: 曲线类型
            rated_params: 额定参数
            constraints: 约束参数
        """
        key = cls._make_key(device_id, curve_type, rated_params)
        cls._cache[key] = constraints
        cls._logger.debug(
            f"约束缓存已设置: device_id={device_id}, curve_type={curve_type}"
        )

    @classmethod
    def invalidate(cls, device_id: int) -> int:
        """
        使指定设备的缓存失效

        Args:
            device_id: 设备ID

        Returns:
            int: 清除的缓存数量
        """
        keys_to_remove = [k for k in cls._cache if k[0] == device_id]
        for key in keys_to_remove:
            del cls._cache[key]

        if keys_to_remove:
            cls._logger.info(
                f"清除设备{device_id}的{len(keys_to_remove)}个约束缓存"
            )

        return len(keys_to_remove)

    @classmethod
    def clear_all(cls) -> int:
        """
        清除所有缓存

        Returns:
            int: 清除的缓存数量
        """
        count = len(cls._cache)
        cls._cache.clear()
        cls._logger.info(f"清除所有约束缓存（共{count}个）")
        return count

    @classmethod
    def _make_key(
        cls,
        device_id: int,
        curve_type: str,
        rated_params: Dict[str, Any]
    ) -> tuple:
        """
        生成缓存键

        Args:
            device_id: 设备ID
            curve_type: 曲线类型
            rated_params: 额定参数

        Returns:
            tuple: 缓存键 (device_id, curve_type, params_hash)
        """
        # 将额定参数序列化并计算哈希
        params_str = json.dumps(rated_params, sort_keys=True)
        params_hash = hashlib.md5(params_str.encode()).hexdigest()
        return (device_id, curve_type, params_hash)

    @classmethod
    def get_stats(cls) -> Dict[str, Any]:
        """
        获取缓存统计

        Returns:
            Dict: 统计信息，包含：
                - hits: 命中次数
                - misses: 未命中次数
                - hit_rate: 命中率
                - cache_size: 缓存大小
        """
        total = cls._stats['hits'] + cls._stats['misses']
        hit_rate = cls._stats['hits'] / total if total > 0 else 0

        return {
            'hits': cls._stats['hits'],
            'misses': cls._stats['misses'],
            'hit_rate': hit_rate,
            'cache_size': len(cls._cache)
        }
```

```python
from typing import Dict, Any, Optional, List
import pandas as pd
```

```python
class SequentialMethodFitter:
    """顺序方法拟合器 - 顺序执行多种拟合方法

    🔴 设计决策：统一使用顺序执行，禁止线程并发
```

```python
import logging
import time
from app.core.logging.setup import log_activity

class SequentialMethodFitter:
    """顺序方法拟合器 - 完整日志示例

    🔴 设计决策：统一使用顺序执行，禁止线程并发
    """

    def __init__(self, method_registry, timeout_seconds=30):
        self._logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")
        self._registry = method_registry
        self._timeout = timeout_seconds

        self._logger.info(
            "[顺序拟合] 初始化",
            extra={"extra_data": {
                "组件": "SequentialMethodFitter",
                "配置": {
                    "超时时间": timeout_seconds,
                    "执行模式": "顺序执行"
                }
            }}
        )

    def fit_sequential(self, methods, X, y, method_params=None):
        """顺序执行多种拟合方法"""

        start_time = time.time()
        method_params = method_params or {}

        with log_activity("顺序拟合", params={
            "methods": methods, "data_points": len(X)
        }):
            self._logger.info(
                "[顺序拟合] 开始顺序拟合",
                extra={"extra_data": {
                    "方法列表": methods,
                    "方法数量": len(methods),
                    "数据点数": len(X),
                    "执行模式": "顺序执行"
                }}
            )

            results = {}
            errors = []

            for i, method_name in enumerate(methods):
                method_start = time.time()
                try:
                    method = self._registry.get(method_name)
                    params = method_params.get(method_name, {})
                    result = self._fit_single(method, X, y, params)
                    results[method_name] = result

                    method_duration = (time.time() - method_start) * 1000
                    self._logger.debug(
                        "[顺序拟合] 方法完成",
                        extra={"extra_data": {
                            "方法名": method_name,
                            "序号": f"{i+1}/{len(methods)}",
                            "R²": result.r_squared,
                            "RMSE": result.rmse,
                            "耗时ms": method_duration
                        }}
                    )

                except Exception as e:
                    errors.append({'method': method_name, 'error': str(e)})
                    self._logger.warning(
                        "[顺序拟合] 方法失败",
                        extra={"extra_data": {
                            "方法名": method_name,
                            "序号": f"{i+1}/{len(methods)}",
                            "错误": str(e)
                        }}
                    )

            total_duration = (time.time() - start_time) * 1000

            self._logger.info(
                "[顺序拟合] 顺序拟合完成",
                extra={"extra_data": {
                    "成功数": len(results),
                    "失败数": len(errors),
                    "成功方法": list(results.keys()),
                    "失败方法": [e['method'] for e in errors],
                    "总耗时ms": total_duration,
                    "平均耗时ms": total_duration / len(methods) if methods else 0
                }}
            )

            return results
```

#### 8.2.1 输入验证装饰器（🟢-04修复）

```python
from typing import Callable, Any, Dict, List, Optional, Union
from functools import wraps
import pandas as pd
import numpy as np
import logging

def validate_input(
    required_params: Optional[List[str]] = None,
    param_types: Optional[Dict[str, type]] = None,
    param_ranges: Optional[Dict[str, tuple]] = None
) -> Callable:
    """
    🟢-04修复：输入验证装饰器

    用于验证函数参数的有效性，减少每个方法中的重复验证代码。

    Args:
        required_params: 必需的参数名列表
        param_types: 参数类型验证 {参数名: 期望类型}
        param_ranges: 参数范围验证 {参数名: (min, max)}

    Example:
        @validate_input(
            required_params=['device_id', 'curve_type'],
            param_types={'device_id': int, 'data': pd.DataFrame},
            param_ranges={'device_id': (1, 10000)}
        )
        def fit(self, device_id, curve_type, data):
            ...
    """
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            logger = logging.getLogger(f"{func.__module__}.{func.__name__}")

            # 获取函数签名，合并args和kwargs
            import inspect
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()
            all_params = bound.arguments

            # 验证必需参数
            if required_params:
                for param in required_params:
                    if param not in all_params or all_params[param] is None:
                        raise ValueError(f"缺少必需参数: {param}")

            # 验证参数类型
            if param_types:
                for param, expected_type in param_types.items():
                    if param in all_params and all_params[param] is not None:
                        if not isinstance(all_params[param], expected_type):
                            actual_type = type(all_params[param]).__name__
                            raise TypeError(
                                f"参数 {param} 类型错误: 期望 {expected_type.__name__}, "
                                f"实际 {actual_type}"
                            )

            # 验证参数范围
            if param_ranges:
                for param, (min_val, max_val) in param_ranges.items():
                    if param in all_params and all_params[param] is not None:
                        val = all_params[param]
                        if val < min_val or val > max_val:
                            raise ValueError(
                                f"参数 {param} 超出范围: {val} 不在 [{min_val}, {max_val}] 内"
                            )

            return func(*args, **kwargs)
        return wrapper
    return decorator


def validate_dataframe(
    required_columns: Optional[List[str]] = None,
    min_rows: int = 1,
    no_nulls: bool = False
) -> Callable:
    """
    🟢-04修复：DataFrame验证装饰器

    Args:
        required_columns: 必需的列名列表
        min_rows: 最小行数
        no_nulls: 是否禁止空值
    """
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 查找DataFrame参数
            for arg in args[1:]:  # 跳过self
                if isinstance(arg, pd.DataFrame):
                    df = arg
                    break
            else:
                for v in kwargs.values():
                    if isinstance(v, pd.DataFrame):
                        df = v
                        break
                else:
                    return func(*args, **kwargs)  # 无DataFrame，跳过验证

            # 验证必需列
            if required_columns:
                missing = set(required_columns) - set(df.columns)
                if missing:
                    raise ValueError(f"DataFrame缺少必需列: {missing}")

            # 验证最小行数
            if len(df) < min_rows:
                raise ValueError(f"DataFrame行数不足: 需要至少{min_rows}行，实际{len(df)}行")

            # 验证空值
            if no_nulls and df.isnull().any().any():
                null_cols = df.columns[df.isnull().any()].tolist()
                raise ValueError(f"DataFrame存在空值，列: {null_cols}")

            return func(*args, **kwargs)
        return wrapper
    return decorator
```

#### 8.2.2 性能监控工具（🟢-07修复）

```python
from typing import Callable, Dict, Any, Optional
from functools import wraps
from dataclasses import dataclass, field
from datetime import datetime
import time
import threading
import tracemalloc
import logging

@dataclass
class PerformanceMetrics:
    """🟢-07修复：性能指标数据类"""
    function_name: str
    start_time: datetime
    end_time: Optional[datetime] = None
    duration_ms: float = 0.0
    memory_before_mb: float = 0.0
    memory_after_mb: float = 0.0
    memory_peak_mb: float = 0.0
    success: bool = True
    error_message: Optional[str] = None
    extra_info: Dict[str, Any] = field(default_factory=dict)


class PerformanceMonitor:
    """
    🟢-07修复：性能监控器（单例模式）

    用于收集拟合过程中的性能指标，支持：
    - 执行时间监控
    - 内存使用监控
    - 聚合统计
    """

    _instance = None
    _lock = threading.Lock()

    def __new__(cls):
        with cls._lock:
            if cls._instance is None:
                cls._instance = super().__new__(cls)
                cls._instance._initialized = False
            return cls._instance

    def __init__(self):
        if self._initialized:
            return
        self._logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")
        self._metrics: List[PerformanceMetrics] = []
        self._aggregated: Dict[str, Dict[str, float]] = {}
        self._initialized = True

    def record(self, metrics: PerformanceMetrics) -> None:
        """记录一次性能指标"""
        with self._lock:
            self._metrics.append(metrics)
            self._update_aggregated(metrics)

    def _update_aggregated(self, metrics: PerformanceMetrics) -> None:
        """更新聚合统计"""
        name = metrics.function_name
        if name not in self._aggregated:
            self._aggregated[name] = {
                'count': 0,
                'total_duration_ms': 0,
                'max_duration_ms': 0,
                'total_memory_mb': 0,
                'max_memory_mb': 0,
                'error_count': 0
            }

        agg = self._aggregated[name]
        agg['count'] += 1
        agg['total_duration_ms'] += metrics.duration_ms
        agg['max_duration_ms'] = max(agg['max_duration_ms'], metrics.duration_ms)
        agg['total_memory_mb'] += metrics.memory_peak_mb
        agg['max_memory_mb'] = max(agg['max_memory_mb'], metrics.memory_peak_mb)
        if not metrics.success:
            agg['error_count'] += 1

    def get_summary(self) -> Dict[str, Dict[str, float]]:
        """获取聚合统计摘要"""
        summary = {}
        for name, agg in self._aggregated.items():
            count = agg['count']
            summary[name] = {
                'avg_duration_ms': agg['total_duration_ms'] / count if count > 0 else 0,
                'max_duration_ms': agg['max_duration_ms'],
                'avg_memory_mb': agg['total_memory_mb'] / count if count > 0 else 0,
                'max_memory_mb': agg['max_memory_mb'],
                'error_rate': agg['error_count'] / count if count > 0 else 0,
                'total_calls': count
            }
        return summary

    def clear(self) -> None:
        """清空所有记录"""
        with self._lock:
            self._metrics.clear()
            self._aggregated.clear()


def monitor_performance(
    track_memory: bool = True,
    log_result: bool = True
) -> Callable:
    """
    🟢-07修复：性能监控装饰器

    Args:
        track_memory: 是否跟踪内存使用
        log_result: 是否记录日志

    Example:
        @monitor_performance(track_memory=True)
        def fit(self, device_id, curve_type, data):
            ...
    """
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            logger = logging.getLogger(f"{func.__module__}.{func.__name__}")
            monitor = PerformanceMonitor()

            metrics = PerformanceMetrics(
                function_name=func.__qualname__,
                start_time=datetime.now()
            )

            # 开始内存跟踪
            if track_memory:
                tracemalloc.start()
                metrics.memory_before_mb = tracemalloc.get_traced_memory()[0] / 1024 / 1024

            start_time = time.time()

            try:
                result = func(*args, **kwargs)
                metrics.success = True
                return result
            except Exception as e:
                metrics.success = False
                metrics.error_message = str(e)
                raise
            finally:
                # 计算耗时
                metrics.end_time = datetime.now()
                metrics.duration_ms = (time.time() - start_time) * 1000

                # 获取内存使用
                if track_memory:
                    current, peak = tracemalloc.get_traced_memory()
                    metrics.memory_after_mb = current / 1024 / 1024
                    metrics.memory_peak_mb = peak / 1024 / 1024
                    tracemalloc.stop()

                # 记录指标
                monitor.record(metrics)

                # 输出日志
                if log_result:
                    logger.info(
                        f"[性能] {func.__qualname__}",
                        extra={"extra_data": {
                            "耗时ms": f"{metrics.duration_ms:.2f}",
                            "内存峰值MB": f"{metrics.memory_peak_mb:.2f}" if track_memory else "N/A",
                            "成功": metrics.success
                        }}
                    )
        return wrapper
    return decorator
```
