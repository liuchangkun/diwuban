#### 3.1.1 创建测试表的setup

**修改文件**：`tests/test_transaction_integration.py`

**修改内容**：

```python
import pytest
from app.core.database import get_db_connection

class TestTransactionIntegration:
    
    @pytest.fixture(autouse=True)
    def setup_test_table(self):
        """创建测试表"""
        with get_db_connection() as conn:
            with conn.cursor() as cursor:
                # 创建测试表
                cursor.execute("""
                    CREATE TABLE IF NOT EXISTS test_table (
                        id SERIAL PRIMARY KEY,
                        name VARCHAR(100) NOT NULL,
                        created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
                    )
                """)
                conn.commit()
        
        yield  # 运行测试
        
        # 清理测试表
        with get_db_connection() as conn:
            with conn.cursor() as cursor:
                cursor.execute("DROP TABLE IF EXISTS test_table")
                conn.commit()
    
    def test_connection_pool_with_transaction(self):
        """测试连接池的事务支持"""
        with get_db_connection() as conn:
            with conn.cursor() as cursor:
                # 插入测试数据
                cursor.execute("INSERT INTO test_table (name) VALUES ('test')")
                conn.commit()
                
                # 验证数据已插入
                cursor.execute("SELECT COUNT(*) FROM test_table WHERE name = 'test'")
                count = cursor.fetchone()[0]
                assert count == 1, "数据应该已插入"
                
                # 回滚测试
                cursor.execute("INSERT INTO test_table (name) VALUES ('rollback_test')")
                conn.rollback()
                
                # 验证数据未插入
                cursor.execute("SELECT COUNT(*) FROM test_table WHERE name = 'rollback_test'")
                count = cursor.fetchone()[0]
                assert count == 0, "回滚后数据不应该存在"
```

**修改说明**：
- 使用`@pytest.fixture(autouse=True)`自动运行setup和teardown
- setup中创建test_table表
- teardown中删除test_table表
- 测试中验证事务的提交和回滚功能
