## 🎯 问题描述

### 问题1.3：设备额定参数完全未使用

**现状**：
- `device_rated_params` 表存在且数据完整（3个设备，每个7个参数）
- `orchestrator.py` 的 `load_parameters()` 方法完全忽略此表
- 所有计算方法都无法使用设备额定参数

**影响**：
- 计算精度降低（所有设备使用相同参数）
- 无法体现设备差异
- 参数优化受限

### 问题2.5：计算参数名不一致

**现状**：
- EFF_SIMPLE_V1：代码使用 `eta_motor_assumed`，数据库存储 `eta_motor`
- pump_head_method_main：代码使用 `P_atm`，数据库存储 `b_H`

**影响**：
- 参数无法从数据库正确读取
- 使用硬编码默认值

### 问题3.7：10个方法缺少数据库参数配置

**现状**：
10个方法在代码中使用参数，但数据库中未配置

**影响**：
- 无法通过数据库调整参数
- 参数优化功能无法使用

### 步骤1：修改 load_parameters() 方法

**目标**：使其能够加载设备额定参数

**文件**：`app/services/calculation/orchestrator.py`

**修改内容**：

```python
def load_parameters(self, device_id: int, method_id: str) -> Dict[str, float]:
    """
    加载计算参数（包括设备额定参数）
    
    参数来源优先级：
    1. 设备级计算参数（calculation_parameters, device_id=X, method_id=Y）
    2. 设备额定参数（device_rated_params, device_id=X）
    3. 全局计算参数（calculation_parameters, device_id=NULL, method_id=Y）
    4. 代码默认值（params.get('xxx', default_value)）
    """
    params = {}
    
    with get_connection() as conn:
        with conn.cursor() as cur:
            # 1. 加载全局计算参数（最低优先级）
            query = """
                SELECT param_name, param_value
                FROM calculation_parameters
                WHERE device_id IS NULL AND method_id = %s
            """
            cur.execute(query, (method_id,))
            for row in cur.fetchall():
                params[row[0]] = float(row[1])
            
            # 2. 加载设备额定参数（新增！）
            query = """
                SELECT param_key, value_numeric
                FROM device_rated_params
                WHERE device_id = %s AND value_numeric IS NOT NULL
            """
            cur.execute(query, (device_id,))
            for row in cur.fetchall():
                params[row[0]] = float(row[1])
            
            # 3. 加载设备级计算参数（最高优先级）
            query = """
                SELECT param_name, param_value
                FROM calculation_parameters
                WHERE device_id = %s AND method_id = %s
            """
            cur.execute(query, (device_id, method_id))
            for row in cur.fetchall():
                params[row[0]] = float(row[1])
    
    logger.debug(
        f"[参数加载] device_id={device_id}, method_id={method_id}, "
        f"params_count={len(params)}, params={params}"
    )
    
    return params
```

```python
def check_rated_params_availability(self, method_id: str, device_id: int) -> bool:
    """检查设备是否有所需的额定参数"""
    REQUIRED_RATED_PARAMS = {
        'pump_speed_method_a': ['rated_frequency', 'poles_pair'],
        'pump_speed_method_b': ['rated_frequency', 'poles_pair'],
        'EFF_SIMPLE_V1': ['rated_efficiency', 'eta_motor'],
    }

    required_params = REQUIRED_RATED_PARAMS.get(method_id, [])
    if not required_params:
        return True

    # 查询设备是否有所需参数
    with get_connection() as conn:
        with conn.cursor() as cur:
            query = """
                SELECT param_key FROM device_rated_params
                WHERE device_id = %s AND param_key = ANY(%s)
                  AND value_numeric IS NOT NULL
            """
            cur.execute(query, (device_id, required_params))
            available_params = {row[0] for row in cur.fetchall()}

    missing_params = set(required_params) - available_params
    if missing_params:
        logger.warning(f"设备{device_id}缺少额定参数: {missing_params}")
        return False

    return True
```

```python
def _build_observation_matrix_with_rated_point(
    self,
    measured_data: Dict[str, np.ndarray],
    calculated_data: Dict[str, np.ndarray],
    rated_params: Dict[str, float],
    method_id: str
) -> Tuple[np.ndarray, np.ndarray]:
    """构建观测矩阵（包含额定工况点）"""

    if method_id == 'HEAD_COEF_V1':
        Q = measured_data['pump_flow_rate']
        H = calculated_data['pump_head']

        # 添加额定工况点（重复10次以增加权重）
        rated_flow = rated_params.get('rated_flow')
        rated_head = rated_params.get('rated_head')

        if rated_flow and rated_head:
            Q_rated = np.full(10, rated_flow)
            H_rated = np.full(10, rated_head)
            Q_all = np.concatenate([Q[valid], Q_rated])
            H_all = np.concatenate([H[valid], H_rated])
        else:
            Q_all = Q[valid]
            H_all = H[valid]

        # 构建观测矩阵
        X = np.column_stack([np.ones(len(Q_all)), Q_all, Q_all**2, ...])
        y = H_all

        return X, y
```

**修改3**：添加额定工况点验证
```python
def _validate_at_rated_point(
    self,
    optimized_params: Dict[str, float],
    rated_params: Dict[str, float],
    method_id: str
) -> float:
    """验证在额定工况点的误差"""

    if method_id == 'HEAD_COEF_V1':
        rated_flow = rated_params.get('rated_flow')
        rated_head = rated_params.get('rated_head')

        if not (rated_flow and rated_head):
            return 0.0

        # 计算在额定流量下的扬程
        H_pred = sum(optimized_params[f'a{i}'] * rated_flow**i for i in range(6))
        error = abs(H_pred - rated_head) / rated_head

        logger.info(
            f"额定工况点验证: Q={rated_flow}, H_pred={H_pred:.2f}, "
            f"H_rated={rated_head:.2f}, error={error:.2%}"
        )

        return error
```

```python
        rated_params=rated_params,  # 传入额定参数！
        device_id=device_id
    )

    # 5. 验证优化效果
    logger.info(
        f"参数优化完成: RMSE={metrics['rmse']:.4f}, "
        f"R²={metrics['r2']:.4f}, confidence={metrics['confidence']:.4f}"
    )

    return optimized_params, metrics
```
