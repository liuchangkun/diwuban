**原则**：
1. **精度优先**：精度高的方法优先级高
2. **依赖可用性**：依赖可用的方法优先级高

**推荐做法**：
- 优先使用数据可用性判断（通过dependencies检查）
- 避免使用设备配置条件（增加复杂度）
- 如果必须使用设备配置条件，需要实现专门的检查逻辑

def calculate_pump_efficiency_eff_curve_v1(
    data: Dict[str, np.ndarray],
    params: Dict[str, float],
    device_id: int = None
) -> np.ndarray:
    """
    基于特性曲线的泵效率计算

    从 pump_characteristic_curves 表中读取效率曲线数据，
    根据实际流量进行线性插值得到效率值。

    公式:
        η = interp(Q, Q_curve, η_curve)

    其中:
        - η: 泵效率（0-1）
        - Q: 实际流量（m³/h）
        - Q_curve: 特性曲线流量点
        - η_curve: 特性曲线效率点
    """
    Q = data['pump_flow_rate']  # m³/h

    # 查询设备的效率特性曲线
    query = """
        SELECT flow_rate, value
        FROM pump_characteristic_curves
        WHERE device_id = %s
          AND curve_type = 'efficiency'
          AND flow_rate IS NOT NULL
          AND value IS NOT NULL
        ORDER BY flow_rate
    """

    # 如果没有特性曲线数据，返回NaN（优雅降级）
    if not curve_data or len(curve_data) < 2:
        logger.warning(f"No efficiency curve data for device {device_id}")
        return np.full_like(Q, np.nan)

    # 线性插值
    efficiency = np.interp(Q, Q_curve, eff_curve, left=np.nan, right=np.nan)

    # 限制效率范围在 [0, 1]
    efficiency = np.clip(efficiency, 0.0, 1.0)

    return efficiency
