#### 步骤2：修改 load_parameters() 方法（完整实现）

```python
def load_parameters(
    self,
    device_id: int,
    method_id: str
) -> Dict[str, float]:
    """
    加载计算参数（4层优先级）
    
    优先级（从低到高）：
    1. 全局计算参数（device_id=NULL, station_id=NULL）
    2. 设备额定参数（device_rated_params表）
    3. 泵站级计算参数（station_id=X, device_id=NULL）
    4. 设备级计算参数（device_id=X）
    
    Args:
        device_id: 设备ID
        method_id: 方法ID
    
    Returns:
        参数字典
    """
    params = {}
    param_sources = {}  # 记录每个参数的来源
    
    with get_connection() as conn:
        with conn.cursor() as cur:
            # 第1层：加载全局计算参数
            query = """
                SELECT param_name, param_value
                FROM calculation_parameters
                WHERE device_id IS NULL AND station_id IS NULL AND method_id = %s
            """
            cur.execute(query, (method_id,))
            for row in cur.fetchall():
                params[row[0]] = float(row[1])
                param_sources[row[0]] = 'global'
            
            # 第2层：加载设备额定参数
            query = """
                SELECT param_key, param_value
                FROM device_rated_params
                WHERE device_id = %s
            """
            cur.execute(query, (device_id,))
            rated_rows = cur.fetchall()
            
            if rated_rows:
                # 设备有额定参数
                for row in rated_rows:
                    params[row[0]] = float(row[1])
                    param_sources[row[0]] = 'rated'
            else:
                # 设备没有额定参数，使用全局默认值
                query = "SELECT param_key, default_value FROM global_default_rated_params"
                cur.execute(query)
                for row in cur.fetchall():
                    params[row[0]] = float(row[1])
                    param_sources[row[0]] = 'global_default_rated'
                logger.warning(
                    f"设备{device_id}没有额定参数，使用全局默认值",
                    extra={"device_id": device_id, "method_id": method_id}
                )
            
            # 第3层：加载泵站级计算参数
            query = """
                SELECT cp.param_name, cp.param_value
                FROM calculation_parameters cp
                JOIN dim_devices d ON cp.station_id = d.station_id
                WHERE d.id = %s AND cp.device_id IS NULL AND cp.method_id = %s
            """
            cur.execute(query, (device_id, method_id))
            for row in cur.fetchall():
                params[row[0]] = float(row[1])
                param_sources[row[0]] = 'station'
            
            # 第4层：加载设备级计算参数（最高优先级）
            query = """
                SELECT param_name, param_value
                FROM calculation_parameters
                WHERE device_id = %s AND method_id = %s
            """
            cur.execute(query, (device_id, method_id))
            for row in cur.fetchall():
                params[row[0]] = float(row[1])
                param_sources[row[0]] = 'device'
    
    # 记录参数来源（用于调试）
    logger.debug(
        f"设备{device_id}方法{method_id}参数加载完成",
        extra={
            "device_id": device_id,
            "method_id": method_id,
            "param_count": len(params),
            "param_sources": param_sources
        }
    )
    
    return params
```

## 问题2：维度表ID变化时的自适应更新机制

### 2.1 问题描述

**核心问题**：`dim_stations`、`dim_devices`、`dim_metric_config` 三个维度表的ID可能发生变化（如重新导入数据），所有依赖这些ID的表需要自动更新。

### 2.2 数据库审查结果

#### 外键约束情况

**查询结果**：23个表有外键约束，但**全部使用 NO ACTION**

| 表名 | 外键字段 | 引用表 | UPDATE规则 | DELETE规则 |
|------|---------|--------|-----------|-----------|
| calculation_parameters | device_id | dim_devices | NO ACTION | NO ACTION |
| calculation_parameters | station_id | dim_stations | NO ACTION | NO ACTION |
| calculation_parameters | metric_key | dim_metric_config | NO ACTION | NO ACTION |
| device_rated_params | device_id | dim_devices | NO ACTION | **CASCADE** |
| completion_runs | device_id | dim_devices | NO ACTION | NO ACTION |
| completion_runs | station_id | dim_stations | NO ACTION | NO ACTION |
| ... | ... | ... | ... | ... |

**问题**：
- ❌ **所有外键的 UPDATE 规则都是 NO ACTION**
- ❌ **如果维度表ID变化，依赖表的ID不会自动更新**
- ❌ **会导致外键约束违反和数据不一致**

#### 包含ID字段但缺少外键约束的表

**查询结果**：大量表包含 device_id/station_id/metric_id 但没有外键约束

**核心业务表**（需要添加外键约束）：
- `calculation_failures_log`
- `calculation_performance_metrics`
- `calculation_validation_config`
- `optimization_history`
- `pump_characteristic_curves`
- `quality_diagnosis_log`
- `quality_profile_log`

**分区表和物化视图**（不需要外键约束）：
- `mpps_w_*`（分区表）
- `mv_*`（物化视图）
- `v_*`（视图）

### 2.3 解决方案

#### 方案A：添加级联更新规则（推荐）

**优点**：
- 自动处理ID变化
- 数据一致性有保障
- 无需业务代码修改

**缺点**：
- 性能开销（级联更新大量数据）
- 需要停机维护（添加外键约束）

**实施步骤**：

**步骤1：为现有外键添加级联更新规则**

3. 泵站级计算参数（station_id=X, device_id=NULL）
4. 设备级计算参数（device_id=X）
5. **优化参数**（optimized_parameters表，最高优先级）

#### 步骤3：修改 optimize_parameters() 方法（保存优化参数）

```python
def optimize_parameters(
    self,
    device_id: int,
    method_id: str,
    measured_data: Dict[str, np.ndarray],
    calculated_data: Dict[str, np.ndarray],
    rated_params: Optional[Dict[str, float]] = None
) -> Dict[str, Any]:
    """
    优化参数（使用RLS算法）
    
    Args:
        device_id: 设备ID
        method_id: 方法ID
        measured_data: 实测数据
        calculated_data: 计算数据
        rated_params: 额定参数（用作约束）
    
    Returns:
        优化结果字典
    """
    # 1. 构建观测矩阵（添加额定工况点约束）
    X, y = self._build_observation_matrix_with_rated_point(
        measured_data, calculated_data, rated_params, method_id
    )
    
    # 2. RLS优化
    theta, P = self._rls_optimize(X, y)
    
    # 3. 计算质量评分
    y_pred = X @ theta
    rmse = np.sqrt(np.mean((y - y_pred) ** 2))
    mae = np.mean(np.abs(y - y_pred))
    r2 = 1 - np.sum((y - y_pred) ** 2) / np.sum((y - np.mean(y)) ** 2)
    mape = np.mean(np.abs((y - y_pred) / y)) * 100
    confidence_score = r2 if r2 > 0 else 0
    
    # 4. 验证在额定工况点的误差
    if rated_params:
        rated_error = self._validate_at_rated_point(
            theta, rated_params, method_id
        )
        if rated_error > 0.05:  # 误差超过5%
            logger.warning(
                f"优化参数在额定工况点误差过大: {rated_error:.2%}",
                extra={"device_id": device_id, "method_id": method_id}
            )
    
    # 5. 保存优化参数到数据库
    self._save_optimized_parameters(
        device_id=device_id,
        method_id=method_id,
        theta=theta,
        rmse=rmse,
        mae=mae,
        r2=r2,
        mape=mape,
        confidence_score=confidence_score
    )
    
    return {
        'optimized_params': theta,
        'metrics': {
            'rmse': rmse,
            'mae': mae,
            'r2': r2,
            'mape': mape,
            'confidence_score': confidence_score
        }
    }

def _save_optimized_parameters(
    self,
    device_id: int,
    method_id: str,
    theta: np.ndarray,
    rmse: float,
    mae: float,
    r2: float,
    mape: float,
    confidence_score: float
):
    """保存优化参数到数据库"""
    with get_connection() as conn:
        with conn.cursor() as cur:
            # 1. 获取参数名列表
            param_names = self._get_param_names(method_id)
            
            # 2. 获取当前最大版本号
            cur.execute(
                """
                SELECT COALESCE(MAX(version), 0) + 1
                FROM optimized_parameters
                WHERE device_id = %s AND method_id = %s
                """,
                (device_id, method_id)
            )
            new_version = cur.fetchone()[0]
            
            # 3. 将旧版本标记为非活跃
            cur.execute(
                """
                UPDATE optimized_parameters
                SET is_active = FALSE
                WHERE device_id = %s AND method_id = %s AND is_active = TRUE
                """,
                (device_id, method_id)
            )
            
            # 4. 插入新版本参数
            for param_name, param_value in zip(param_names, theta):
                cur.execute(
                    """
                    INSERT INTO optimized_parameters (
                        device_id, method_id, param_name, param_value,
                        rmse, mae, r2, mape, confidence_score,
                        version, is_active
                    ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, TRUE)
                    """,
                    (device_id, method_id, param_name, float(param_value),
                     rmse, mae, r2, mape, confidence_score, new_version)
                )
            
            conn.commit()
            
            logger.info(
                f"优化参数已保存（版本{new_version}）",
                extra={
                    "device_id": device_id,
                    "method_id": method_id,
                    "version": new_version,
                    "r2": r2
                }
            )
```
