```python
class DatabaseValidator:
    """数据库验证工具"""
    
    def __init__(self, settings):
        self.settings = settings
        self.conn = get_conn(settings)
    
    def query_and_verify(self, sql: str, expected: dict):
        """执行查询并验证结果"""
        result = self.conn.execute(sql).fetchall()
        assert result == expected, f"Expected {expected}, got {result}"
    
    def verify_row_count(self, table: str, expected_count: int):
        """验证表行数"""
        sql = f"SELECT COUNT(*) FROM {table}"
        result = self.conn.execute(sql).fetchone()[0]
        assert result == expected_count
    
    def verify_no_duplicates(self, table: str, columns: list):
        """验证无重复数据"""
        cols = ", ".join(columns)
        sql = f"""
            SELECT COUNT(*) FROM (
                SELECT {cols}, COUNT(*) as cnt 
                FROM {table} 
                GROUP BY {cols} 
                HAVING cnt > 1
            ) t
        """
        result = self.conn.execute(sql).fetchone()[0]
        assert result == 0, f"Found {result} duplicate rows"
```

```python
class FileValidator:
    """文件系统验证工具"""
    
    @staticmethod
    def verify_file_exists(file_path: Path):
        """验证文件存在"""
        assert file_path.exists(), f"File {file_path} does not exist"
    
    @staticmethod
    def verify_file_content(file_path: Path, expected_content: str):
        """验证文件内容"""
        with open(file_path, 'r') as f:
            content = f.read()
        assert content == expected_content
    
    @staticmethod
    def verify_file_size(file_path: Path, min_size: int, max_size: int):
        """验证文件大小"""
        size = file_path.stat().st_size
        assert min_size <= size <= max_size
    
    @staticmethod
    def verify_csv_structure(file_path: Path, expected_columns: list):
        """验证CSV文件结构"""
        import csv
        with open(file_path, 'r') as f:
            reader = csv.DictReader(f)
            assert reader.fieldnames == expected_columns
```

```python
class LogValidator:
    """日志验证工具"""
    
    def __init__(self, log_file: Path):
        self.log_file = log_file
    
    def verify_log_contains(self, pattern: str):
        """验证日志包含特定内容"""
        with open(self.log_file, 'r') as f:
            content = f.read()
        assert pattern in content, f"Pattern '{pattern}' not found in logs"
    
    def verify_log_count(self, pattern: str, expected_count: int):
        """验证日志出现次数"""
        with open(self.log_file, 'r') as f:
            content = f.read()
        count = content.count(pattern)
        assert count == expected_count
    
    def verify_error_log(self, error_pattern: str):
        """验证错误日志"""
        with open(self.log_file, 'r') as f:
            for line in f:
                if 'ERROR' in line and error_pattern in line:
                    return True
        assert False, f"Error pattern '{error_pattern}' not found in logs"
```

### 5. 性能监控工具

**文件**: `tests/utils/performance_monitor.py`

```python
class PerformanceMonitor:
    """性能监控工具"""
    
    def __init__(self):
        self.metrics = {}
    
    def measure_time(self, operation_name: str):
        """测量操作耗时"""
        import time
        start = time.time()
        yield
        duration = time.time() - start
        self.metrics[operation_name] = duration
    
    def measure_memory(self, operation_name: str):
        """测量内存使用"""
        import psutil
        process = psutil.Process()
        start_memory = process.memory_info().rss
        yield
        end_memory = process.memory_info().rss
        self.metrics[f"{operation_name}_memory"] = end_memory - start_memory
    
    def verify_performance(self, operation_name: str, max_duration: float):
        """验证性能指标"""
        duration = self.metrics.get(operation_name)
        assert duration <= max_duration, \
            f"Operation {operation_name} took {duration}s, max {max_duration}s"
```
